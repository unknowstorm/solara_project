private val SpMiniInset = 8.dp

/** Правила Spotify: углы обложки 4 dp, без изменений формы. */

/** Общее состояние мини-плеера Spotify. Живёт только в памяти. */
@Stable
class SpotifyMiniState {
    /** Мини-плеер сейчас должен быть виден (трек есть, наш плеер не играет). */
    var shown by mutableStateOf(false)
    var expanded by mutableStateOf(false)
    /** Трек, который пользователь закрыл крестиком или свайпом. */
    var dismissedUri by mutableStateOf<String?>(null)
    val anchor = MiniPlayerAnchor()
}

val LocalSpotifyMini = staticCompositionLocalOf { SpotifyMiniState() }

/** Пустое место в MainScreen, где стоит мини-плеер Spotify (рисует его SpotifyExpandablePlayer). */
@Composable
fun SpotifyMiniSlot(hidden: Boolean = false) {
    val ui = LocalSpotifyMini.current
    if (!ui.shown || hidden) return
    Spacer(
        Modifier
            .padding(horizontal = 8.dp, vertical = 8.dp)
            .fillMaxWidth()
            .height(MiniHeight)
            .onGloballyPositioned { ui.anchor.bounds = it.boundsInRoot() }
    )
}

/**
 * Мини-плеер и полный плеер Spotify: тот же морфинг, что у ExpandablePlayer,
 * но отдельный компонент: ничего общего с AudioBackend и нашими источниками.
 * Ставить в корневой Box поверх основного контента.
 */
@Composable
fun SpotifyExpandablePlayer(remote: SpotifyRemote, audio: AudioBackend, modifier: Modifier = Modifier) {
    val ui = LocalSpotifyMini.current
    val anchor = ui.anchor
    val ownAnchor = LocalMiniPlayerAnchor.current
    val stateHolder = remote.state.collectAsState()
    val audioHolder = audio.state.collectAsState()
    // Сырое условие. Spotify при смене трека может на секунду прислать «пустое» состояние
    val rawHasTrack by remember {
        derivedStateOf {
            val st = stateHolder.value
            st.connected && st.premium != false && st.title != null &&
                audioHolder.value.title == null &&
                !(ui.dismissedUri != null && ui.dismissedUri == st.uri && st.isPaused)
        }
    }
    val audioBusy by remember { derivedStateOf { audioHolder.value.title != null } }
    // Последнее состояние с названием трека: его показываем, пока длится короткий разрыв
    val held = remember { mutableStateOf(stateHolder.value) }
    LaunchedEffect(Unit) {
        snapshotFlow { stateHolder.value }.collect { if (it.title != null) held.value = it }
    }
    // Spotify на миг присылает пустое состояние при смене трека: держим плеер до 1,2 с, но только в неявных случаях
    var everShown by remember { mutableStateOf(false) }
    var graceExpired by remember { mutableStateOf(false) }
    val explicitHide by remember {
        derivedStateOf {
            val st = stateHolder.value
            // Наш плеер должен сначала занять своё место, и только потом Spotify скрывается: без разрыва между ними
            (audioBusy && ownAnchor.bounds != null) || !st.connected || st.premium == false ||
                (ui.dismissedUri != null && ui.dismissedUri == st.uri && st.isPaused)
        }
    }
    LaunchedEffect(rawHasTrack) {
        if (rawHasTrack) {
            everShown = true
            graceExpired = false
        } else {
            delay(1200)
            graceExpired = true
        }
    }
    // Считаем прямо в композиции: плеер скрывается и появляется в том же кадре, что и смена источника
    val hasTrack = rawHasTrack || (everShown && !graceExpired && !explicitHide)
    val expanded = ui.expanded
    val scope = rememberCoroutineScope()

    SideEffect { ui.shown = hasTrack }
    LaunchedEffect(Unit) {
        snapshotFlow { Triple(rawHasTrack, audioHolder.value.title != null, stateHolder.value.title) }.collect {
            println("Solara: timeline spotifyMini=${it.first} audioTitle=${it.second} spTitle=${it.third} spPaused=${stateHolder.value.isPaused} audioPlaying=${audioHolder.value.isPlaying}")
        }
    }
    DisposableEffect(Unit) { onDispose { ui.shown = false } }
    // Снова нажали play: закрытый ранее трек опять показываем
    LaunchedEffect(Unit) {
        snapshotFlow { stateHolder.value.isPaused }.collect { paused -> if (!paused) ui.dismissedUri = null }
    }

    val progress = remember { mutableFloatStateOf(if (expanded) 1f else 0f) }
    var animJob by remember { mutableStateOf<Job?>(null) }
    val settling = remember { floatArrayOf(-1f) }

    fun settle(target: Float, velocity: Float = 0f) {
        settling[0] = target
        animJob?.cancel()
        animJob = scope.launch {
            animate(
                initialValue = progress.floatValue,
                targetValue = target,
                initialVelocity = velocity,
                animationSpec = spring(
                    dampingRatio = Spring.DampingRatioNoBouncy,
                    stiffness = Spring.StiffnessMediumLow,
                ),
            ) { v, _ -> progress.floatValue = v.coerceIn(0f, 1f) }
            settling[0] = -1f
        }
    }

    LaunchedEffect(expanded) {
        val target = if (expanded) 1f else 0f
        if (settling[0] != target && progress.floatValue != target) settle(target)
    }

    LaunchedEffect(hasTrack) {
        if (!hasTrack) {
            animJob?.cancel()
            settling[0] = -1f
            progress.floatValue = 0f
            ui.expanded = false
        }
    }
    if (!hasTrack) return

    PlatformBackHandler(enabled = expanded, onBack = { ui.expanded = false })

    var origin by remember { mutableStateOf(Offset.Zero) }
    var miniOffsetX by remember { mutableFloatStateOf(0f) }
    var dismissJob by remember { mutableStateOf<Job?>(null) }

    fun animateMiniX(to: Float, after: () -> Unit = {}) {
        dismissJob?.cancel()
        dismissJob = scope.launch {
            animate(initialValue = miniOffsetX, targetValue = to) { v, _ -> miniOffsetX = v }
            after()
        }
    }

    fun dismissMini() {
        ui.dismissedUri = stateHolder.value.uri
        remote.pause()
    }

    BoxWithConstraints(
        modifier
            .fillMaxSize()
            .onGloballyPositioned { origin = it.positionInRoot() }
    ) {
        val density = LocalDensity.current
        val fullW = constraints.maxWidth.toFloat()
        val fullH = constraints.maxHeight.toFloat()
        val fullWidthDp = maxWidth
        val fullHeightDp = maxHeight

        val topInset = WindowInsets.statusBars.asPaddingValues().calculateTopPadding()
        val bottomInset = WindowInsets.navigationBars.asPaddingValues().calculateBottomPadding()

        val fbMargin = with(density) { 8.dp.toPx() }
        val fbBottom = with(density) { (bottomInset + 8.dp).toPx() }
        val miniHpx = with(density) { MiniHeight.toPx() }
        fun miniRect(): Rect {
            val b = anchor.bounds
            return if (b != null) {
                Rect(b.left - origin.x, b.top - origin.y, b.right - origin.x, b.bottom - origin.y)
            } else {
                Rect(fbMargin, fullH - fbBottom - miniHpx, fullW - fbMargin, fullH - fbBottom)
            }
        }
        val miniNow = miniRect()
        val miniWidthDp = with(density) { miniNow.width.toDp() }

        val layoutDirection = LocalLayoutDirection.current
        val sideInsets = WindowInsets.systemBars.union(WindowInsets.displayCutout)
        val leftInset = with(density) { sideInsets.getLeft(density, layoutDirection).toDp() }
        val rightInset = with(density) { sideInsets.getRight(density, layoutDirection).toDp() }
        val landscape = fullW > fullH

        val coverTop: Dp
        val coverSize: Dp
        val coverLeft: Dp
        if (landscape) {
            val availH = maxHeight - topInset - bottomInset
            coverSize = maxOf(120.dp, minOf(availH - 24.dp, maxWidth * 0.42f))
            coverTop = topInset + (availH - coverSize) / 2f
            coverLeft = leftInset + 32.dp
        } else {
            coverTop = topInset + 32.dp + SpCoverDownShift
            coverSize = maxOf(160.dp, minOf(maxWidth - SpScreenSideMargin * 2f, maxHeight - coverTop - bottomInset - SpBelowCoverBlock))
            coverLeft = (maxWidth - coverSize) / 2f
        }
        val coverTopPx = with(density) { coverTop.toPx() }
        val bigX = with(density) { coverLeft.toPx() }
        val belowCoverStart = coverTop + coverSize + 20.dp
        val panelStart = coverLeft + coverSize + 32.dp

        val miniCornerPx = with(density) { SpMiniCorner.toPx() }
        val shadowPx = with(density) { 3.dp.toPx() }
        val miniColor = MaterialTheme.colorScheme.secondaryContainer
        val fullColor = MaterialTheme.colorScheme.surface

        // Запоминаем в момент расстановки, стоит ли плеер на измеренном месте: до этого он невидим
        val placedOnAnchor = remember { mutableStateOf(false) }
        val miniVisible by remember { derivedStateOf { progress.floatValue < 0.999f } }
        val fullVisible by remember { derivedStateOf { progress.floatValue > 0.001f } }

        val dragState = rememberDraggableState { delta ->
            val travel = miniRect().top.coerceAtLeast(1f)
            progress.floatValue = (progress.floatValue - delta / travel).coerceIn(0f, 1f)
        }

        Box(
            modifier = Modifier
                .offset {
                    val m = miniRect()
                    val p = progress.floatValue
                    IntOffset(lerp(m.left, 0f, p).roundToInt(), lerp(m.top, 0f, p).roundToInt())
                }
                .layout { measurable, _ ->
                    placedOnAnchor.value = anchor.bounds != null
                    val m = miniRect()
                    val p = progress.floatValue
                    val w = lerp(m.width, fullW, p).roundToInt().coerceAtLeast(0)
                    val h = lerp(m.height, fullH, p).roundToInt().coerceAtLeast(0)
                    val placeable = measurable.measure(Constraints.fixed(w, h))
                    layout(w, h) { placeable.place(0, 0) }
                }
                .graphicsLayer {
                    val p = progress.floatValue
                    translationX = miniOffsetX
                    // Пока место под мини-плеер не измерено, не рисуем его: иначе он мелькает поверх нижней панели
                    alpha = if (!placedOnAnchor.value && p < 0.01f) 0f else 1f
                    shape = RoundedCornerShape(CornerSize(lerp(miniCornerPx, 0f, p)))
                    clip = true
                    shadowElevation = lerp(shadowPx, 0f, p)
                }
                .drawBehind { drawRect(lerpColor(miniColor, fullColor, progress.floatValue)) }
                .draggable(
                    state = dragState,
                    orientation = Orientation.Vertical,
                    onDragStarted = {
                        animJob?.cancel()
                        settling[0] = -1f
                    },
                    onDragStopped = { velocity ->
                        val p = progress.floatValue
                        val travel = miniRect().top.coerceAtLeast(1f)
                        val target = when {
                            velocity > 1500f -> 0f
                            velocity < -1500f -> 1f
                            p < 0.5f -> 0f
                            else -> 1f
                        }
                        settle(target, -velocity / travel)
                        ui.expanded = target == 1f
                    },
                )
                .then(
                    if (expanded) {
                        Modifier
                    } else {
                        Modifier
                            .pointerInput(Unit) {
                                detectHorizontalDragGestures(
                                    onDragStart = { dismissJob?.cancel() },
                                    onDragEnd = {
                                        val m = miniRect()
                                        if (abs(miniOffsetX) > m.width * 0.35f) {
                                            animateMiniX(if (miniOffsetX > 0f) fullW else -fullW) {
                                                dismissMini()
                                                miniOffsetX = 0f
                                            }
                                        } else {
                                            animateMiniX(0f)
                                        }
                                    },
                                    onDragCancel = { animateMiniX(0f) },
                                    onHorizontalDrag = { _, dx -> miniOffsetX += dx },
                                )
                            }
                            .clickable { ui.expanded = true }
                    }
                )
        ) {
            if (miniVisible) {
                SpMiniContent(remote, held, { progress.floatValue }, miniWidthDp) { dismissMini() }
            }
            if (fullVisible) {
                SpFullContent(
                    remote = remote,
                    stateHolder = held,
                    expandProgress = { progress.floatValue },
                    width = fullWidthDp,
                    height = fullHeightDp,
                    topInset = topInset,
                    bottomInset = bottomInset,
                    belowCoverStart = belowCoverStart,
                    landscape = landscape,
                    panelStart = panelStart,
                    rightInset = rightInset,
                )
            }
            SpSharedCover(
                stateHolder = held,
                progress = { progress.floatValue },
                coverSize = coverSize,
                bigX = bigX,
                bigY = coverTopPx,
            )
        }
    }
}

@Composable
private fun SpMiniContent(
    remote: SpotifyRemote,
    stateHolder: State<SpotifyState>,
    progress: () -> Float,
    width: Dp,
    onClose: () -> Unit,
) {
    val s by stateHolder
    val title = s.title ?: return

    CompositionLocalProvider(LocalContentColor provides MaterialTheme.colorScheme.onSecondaryContainer) {
        Row(
            Modifier
                .wrapContentSize(Alignment.TopStart, unbounded = true)
                .size(width = width, height = MiniHeight)
                .graphicsLayer { alpha = (1f - progress() / 0.25f).coerceIn(0f, 1f) }
                .padding(SpMiniInset),
            verticalAlignment = Alignment.CenterVertically,
        ) {
            Spacer(Modifier.size(SpMiniCoverSize))
            Spacer(Modifier.width(12.dp))
            Column(Modifier.weight(1f)) {
                Text(title, style = MaterialTheme.typography.titleMedium, maxLines = 1, overflow = TextOverflow.Ellipsis)
                s.artist?.let {
                    Text(it, style = MaterialTheme.typography.bodyMedium, maxLines = 1, overflow = TextOverflow.Ellipsis)
                }
            }
            Spacer(Modifier.width(6.dp))
            SpotifyIconChip(24.dp)
            Spacer(Modifier.width(8.dp))
            IconButton(
                onClick = { if (s.isPaused) remote.resume() else remote.pause() },
                modifier = Modifier.size(40.dp).clip(CircleShape).background(MaterialTheme.colorScheme.primary),
            ) {
                PlayPauseIcon(!s.isPaused, Modifier.size(22.dp), MaterialTheme.colorScheme.onPrimary)
            }
            Spacer(Modifier.width(8.dp))
            IconButton(onClick = onClose, modifier = Modifier.size(40.dp)) {
                CloseIcon(Modifier.size(20.dp), MaterialTheme.colorScheme.onSurfaceVariant)
            }
        }
    }
}

/** Общая обложка: морфинг 48 dp -> большая. Без наклона, блика, значков и блюра, углы 4 dp, без обрезки. */
@Composable
private fun SpSharedCover(
    stateHolder: State<SpotifyState>,
    progress: () -> Float,
    coverSize: Dp,
    bigX: Float,
    bigY: Float,
) {
    val density = LocalDensity.current
    val cover by remember { derivedStateOf { stateHolder.value.cover } }

    val sizePx = with(density) { coverSize.toPx() }
    val miniSizePx = with(density) { SpMiniCoverSize.toPx() }
    val miniXY = with(density) { SpMiniInset.toPx() }
    val cornerPx = with(density) { SpCoverCorner.toPx() }

    Box(
        Modifier
            .wrapContentSize(Alignment.TopStart, unbounded = true)
            .size(coverSize)
            .graphicsLayer {
                val p = progress()
                val sc = lerp(miniSizePx / sizePx, 1f, p)
                transformOrigin = TransformOrigin(0f, 0f)
                scaleX = sc
                scaleY = sc
                translationX = lerp(miniXY, bigX, p)
                translationY = lerp(miniXY, bigY, p)
            }
    ) {
        Box(
            Modifier
                .fillMaxSize()
                .graphicsLayer {
                    // 4 dp на экране при любом масштабе
                    val sc = lerp(miniSizePx / sizePx, 1f, progress())
                    shape = RoundedCornerShape(CornerSize(cornerPx / sc))
                    clip = true
                }
                .background(MaterialTheme.colorScheme.surfaceVariant)
        ) {
            cover?.let { Image(it, null, Modifier.fillMaxSize(), contentScale = ContentScale.Fit) }
        }
    }
}

@Composable
private fun SpFullContent(
    remote: SpotifyRemote,
    stateHolder: State<SpotifyState>,
    expandProgress: () -> Float,
    width: Dp,
    height: Dp,
    topInset: Dp,
    bottomInset: Dp,
    belowCoverStart: Dp,
    landscape: Boolean,
    panelStart: Dp,
    rightInset: Dp,
) {
    val s by stateHolder
    var dragValue by remember { mutableStateOf<Float?>(null) }
    var showDetails by remember { mutableStateOf(false) }
    // Сообщение при лайке по правилам Spotify; пропадает само
    var likeNote by remember { mutableStateOf<String?>(null) }
    LaunchedEffect(likeNote) { if (likeNote != null) { delay(2000); likeNote = null } }

    val trackProgress = if (s.durationMs > 0) (s.positionMs.toFloat() / s.durationMs).coerceIn(0f, 1f) else 0f
    val shownProgress = dragValue ?: trackProgress
    val shownPositionMs = dragValue?.let { (it * s.durationMs).toLong() } ?: s.positionMs
    val active = MaterialTheme.colorScheme.primary
    val inactive = MaterialTheme.colorScheme.onSurfaceVariant

    CompositionLocalProvider(LocalContentColor provides MaterialTheme.colorScheme.onSurface) {
        Box(
            Modifier
                .wrapContentSize(Alignment.TopStart, unbounded = true)
                .size(width = width, height = height)
                .graphicsLayer { alpha = ((expandProgress() - 0.3f) / 0.5f).coerceIn(0f, 1f) }
        ) {
            Box(
                Modifier
                    .align(Alignment.TopCenter)
                    .padding(top = topInset + 8.dp)
                    .width(40.dp)
                    .height(4.dp)
                    .clip(CircleShape)
                    .background(MaterialTheme.colorScheme.onSurfaceVariant.copy(alpha = 0.3f))
            )
            // Логотип Spotify на экране с его контентом
            SpotifyLogoChip(24.dp, Modifier.align(Alignment.TopCenter).padding(top = topInset + 20.dp))

            Column(
                modifier = if (landscape) {
                    Modifier
                        .fillMaxSize()
                        .padding(start = panelStart, end = rightInset + 32.dp, top = topInset + 16.dp, bottom = bottomInset + 8.dp)
                } else {
                    Modifier.fillMaxSize().padding(horizontal = SpScreenSideMargin).padding(bottom = bottomInset)
                },
                verticalArrangement = if (landscape) Arrangement.Center else Arrangement.Top,
            ) {
                if (!landscape) Spacer(Modifier.height(belowCoverStart))

                Text(
                    s.title ?: "",
                    style = if (landscape) MaterialTheme.typography.titleLarge else MaterialTheme.typography.headlineSmall,
                    maxLines = if (landscape) 1 else 2,
                    overflow = TextOverflow.Ellipsis,
                    modifier = Modifier.fillMaxWidth(),
                )
                Text(
                    s.artist ?: "",
                    style = MaterialTheme.typography.bodyLarge,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis,
                    modifier = Modifier.fillMaxWidth(),
                )
                s.error?.let {

/** Запрос подборки с повторами: на холодном старте сеть часто ещё не готова (DNS, таймаут). */
private suspend fun searchWithRetry(provider: MusicProvider, query: String): List<Track> {
    var last: Exception? = null
    repeat(3) { attempt ->
        try {
            return provider.search(query)
        } catch (e: CancellationException) {
            throw e
        } catch (e: Exception) {
            last = e
            delay(800L * (attempt + 1))
        }
    }
    throw last ?: IllegalStateException("search failed")
}

/** Лента одного источника. Живёт на уровне MainScreen, чтобы не грузиться заново при смене вкладок. */
class HomeFeed(
    val service: Service,
    private val provider: MusicProvider,
    private val mixes: List<Pair<String, String>>,
) {
    var sections by mutableStateOf<List<FeedSection>>(emptyList())
        private set
    var loading by mutableStateOf(false)
        private set
    var error by mutableStateOf<String?>(null)
        private set

    suspend fun loadIfNeeded() {
        if (sections.isEmpty() && !loading) {
            load()
            // Если сеть так и не ответила, пробуем ещё пару раз сами, без нажатия «Повторить»
            var tries = 0
            while (sections.isEmpty() && error != null && tries < 2) {
                tries++
                delay(4000)
                load()
            }
        }
    }

    suspend fun load() {
        loading = true
        error = null
        try {
            var lastError: String? = null
            val loaded = coroutineScope {
                mixes.map { (title, query) ->
                    async {
                        try {
                            FeedSection(title, searchWithRetry(provider, query).take(12))
                        } catch (e: CancellationException) {
                            throw e
                        } catch (e: Exception) {
                            println("Solara: feed ${service.title}/$title failed: ${e.stackTraceToString()}")
                            lastError = e.message
                            null
                        }
                    }
                }.awaitAll()
            }.filterNotNull().filter { it.tracks.isNotEmpty() }
            sections = loaded
            if (loaded.isEmpty() && lastError != null) error = "Ошибка загрузки: $lastError"
        } finally {
            loading = false
        }
    }
}

private data class FeedRow(val key: String, val title: String, val service: Service?, val tracks: List<Track>)

private fun <T> roundRobin(lists: List<List<T>>): List<T> {
    val result = mutableListOf<T>()
    val longest = lists.maxOfOrNull { it.size } ?: 0
    for (i in 0 until longest) {
        for (list in lists) list.getOrNull(i)?.let(result::add)
    }
    return result
}

/** Собирает ряды главной. На вкладке «Все» источники чередуются, а сверху идёт общий микс. */
private fun buildRows(feeds: List<HomeFeed>, mixed: Boolean): List<FeedRow> {
    val perFeed = feeds.map { feed ->
        feed.sections.map { FeedRow("${feed.service.name}:${it.title}", it.title, feed.service, it.tracks.filterNot { t -> DeadTracks.isDead(t) }) }
    }
    val rows = roundRobin(perFeed)
    if (!mixed) return rows
    val firstRows = perFeed.mapNotNull { it.firstOrNull() }
    if (firstRows.size < 2) return rows
    val mix = roundRobin(firstRows.map { it.tracks }).take(16)
    return listOf(FeedRow("mix", "Микс из всех источников", Service.ALL, mix)) + rows
}

/**
 * Лайки хранят только метаданные, поэтому ссылку на звук берём заново у источника:
 * у Deezer она протухает. Что не удалось получить, пропускаем.
 */
internal suspend fun playLiked(
    tracks: List<Track>,
    start: Track,
    providers: List<MusicProvider>,
    audio: AudioBackend,
) {
    // Окно вокруг выбранного трека: так работает и выбор из хвоста длинного списка
    val from = tracks.indexOfFirst { it.provider == start.provider && it.id == start.id }.coerceAtLeast(0)
    val window = tracks.drop((from - 5).coerceAtLeast(0)).take(30)
    val resolved = coroutineScope {
        window.map { track ->
            async {
                val provider = providers.firstOrNull { it.id == track.provider } ?: return@async null
                val url = try {
                    withTimeoutOrNull(6_000) { provider.resolveStream(track) }
                } catch (e: CancellationException) {
                    throw e
                } catch (e: Exception) {
                    null
                }
                url?.takeIf { it.isNotBlank() }?.let { track.copy(streamUrl = it) }
            }
        }.awaitAll()
    }.filterNotNull()
    val index = resolved.indexOfFirst { it.provider == start.provider && it.id == start.id }
    if (index >= 0) audio.playQueue(resolved, index)
}

package com.unknowstorm.musicapp.ui

import com.unknowstorm.musicapp.model.Track
import com.unknowstorm.musicapp.provider.MusicProvider
import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.async
import kotlinx.coroutines.awaitAll
import kotlinx.coroutines.coroutineScope

/** Ищет во всех источниках сразу и перемешивает результаты. Ошибка одного источника не ломает остальные. */
suspend fun searchAll(providers: List<MusicProvider>, query: String): List<Track> {
    var firstError: Exception? = null
    var failures = 0
    val lists = coroutineScope {
        providers.map { provider ->
            async {
                try {
                    provider.search(query)
                } catch (e: CancellationException) {
                    throw e
                } catch (e: Exception) {
                    println("Solara: ${provider.id} search failed: ${e.message}")
                    failures++
                    firstError = firstError ?: e
                    emptyList<Track>()
                }
            }
        }.awaitAll()
    }
    if (providers.isNotEmpty() && failures == providers.size) {
        throw firstError ?: IllegalStateException("Источники недоступны")
    }
    return interleave(lists)
}

private fun <T> interleave(lists: List<List<T>>): List<T> {
    val result = mutableListOf<T>()
    val longest = lists.maxOfOrNull { it.size } ?: 0
    for (i in 0 until longest) {
        for (list in lists) list.getOrNull(i)?.let(result::add)
    }
    return result
}

 *  1) зарезервировать в MainScreen место под мини-плеер (как раньше, высота 64dp);
 *  2) сообщить [ExpandablePlayer], где именно это место находится.

 * Сигнатура осталась прежней, поэтому MainScreen менять не нужно.
 */
@Composable

    audio: AudioBackend,
    @Suppress("UNUSED_PARAMETER") onOpen: () -> Unit,
    modifier: Modifier = Modifier,
) {
    val anchor = LocalMiniPlayerAnchor.current
    val playerState = audio.state.collectAsState()
    val hasTrack by remember { derivedStateOf { playerState.value.title != null } }
    if (!hasTrack) return

    DisposableEffect(anchor) { onDispose { anchor.bounds = null } }

    Box(
        modifier
            .fillMaxWidth()
            .height(MiniHeight)
            .onGloballyPositioned { anchor.bounds = it.boundsInRoot() }
    )
}

    val hasTrack by remember { derivedStateOf { playerState.value.title != null } }

    val scope = rememberCoroutineScope()

    // Единственный источник правды для анимации: 0 = мини-плеер, 1 = полный экран.
    // Читаем его ТОЛЬКО внутри lambda (layout / graphicsLayer / draw), поэтому кадры анимации
    // не вызывают рекомпозицию.
    val progress = remember { mutableFloatStateOf(if (expanded) 1f else 0f) }
    var animJob by remember { mutableStateOf<Job?>(null) }
    val settling = remember { floatArrayOf(-1f) }

    fun settle(target: Float, velocity: Float = 0f) {
        settling[0] = target
        animJob?.cancel()
        animJob = scope.launch {
            animate(
                initialValue = progress.floatValue,
                targetValue = target,
                initialVelocity = velocity,
                animationSpec = spring(
                    dampingRatio = Spring.DampingRatioNoBouncy,
                    stiffness = Spring.StiffnessMediumLow,
                ),
            ) { v, _ -> progress.floatValue = v.coerceIn(0f, 1f) }
            settling[0] = -1f
        }
    }

    // Родитель поменял expanded (тап по мини-плееру, кнопка «назад») — запускаем анимацию.
    // Если анимацию уже запустил жест (с учётом скорости), второй раз не стартуем.
    LaunchedEffect(expanded) {
        val target = if (expanded) 1f else 0f
        if (settling[0] != target && progress.floatValue != target) settle(target)
    }

    // Трек закончился или нажали ✕: всё прячем
    LaunchedEffect(hasTrack) {
        if (!hasTrack) {
            animJob?.cancel()
            settling[0] = -1f
            progress.floatValue = 0f
            onChange(false)
        }
    }
    if (!hasTrack) return

    PlatformBackHandler(enabled = expanded, onBack = { onChange(false) })

    var origin by remember { mutableStateOf(Offset.Zero) }
    var miniOffsetX by remember { mutableFloatStateOf(0f) }
    var dismissJob by remember { mutableStateOf<Job?>(null) }

    fun animateMiniX(to: Float, after: () -> Unit = {}) {
        dismissJob?.cancel()
        dismissJob = scope.launch {
            animate(initialValue = miniOffsetX, targetValue = to) { v, _ -> miniOffsetX = v }
            after()
        }
    }

    BoxWithConstraints(
        modifier
            .fillMaxSize()
            .onGloballyPositioned { origin = it.positionInRoot() }
    ) {
        val density = LocalDensity.current
        val fullW = constraints.maxWidth.toFloat()
        val fullH = constraints.maxHeight.toFloat()
        val fullWidthDp = maxWidth
        val fullHeightDp = maxHeight

        val topInset = WindowInsets.statusBars.asPaddingValues().calculateTopPadding()
        val bottomInset = WindowInsets.navigationBars.asPaddingValues().calculateBottomPadding()

        // --- Где стоит мини-плеер (в координатах этого оверлея) ---
        val fbMargin = with(density) { MiniSideMargin.toPx() }
        val fbBottom = with(density) { (bottomInset + 8.dp).toPx() }
        val miniHpx = with(density) { MiniHeight.toPx() }
        fun miniRect(): Rect {
            val b = anchor.bounds
            return if (b != null) {
                Rect(b.left - origin.x, b.top - origin.y, b.right - origin.x, b.bottom - origin.y)
            } else {
                // Запасной вариант, пока MainScreen не сообщил положение
                Rect(fbMargin, fullH - fbBottom - miniHpx, fullW - fbMargin, fullH - fbBottom)
            }
        }
        val miniNow = miniRect()
        val miniWidthDp = with(density) { miniNow.width.toDp() }

        // --- Геометрия полного экрана (в dp) ---
        val layoutDirection = LocalLayoutDirection.current
        val sideInsets = WindowInsets.systemBars.union(WindowInsets.displayCutout)
        val leftInset = with(density) { sideInsets.getLeft(density, layoutDirection).toDp() }
        val rightInset = with(density) { sideInsets.getRight(density, layoutDirection).toDp() }
        val landscape = fullW > fullH

        val coverTop: Dp
        val coverSize: Dp
        val coverLeft: Dp
        if (landscape) {
            // Альбомная ориентация: обложка слева по центру, справа название и кнопки управления
            val availH = maxHeight - topInset - bottomInset
            coverSize = maxOf(120.dp, minOf(availH - 24.dp, maxWidth * 0.42f))
            coverTop = topInset + (availH - coverSize) / 2f
            coverLeft = leftInset + 32.dp
        } else {
            coverTop = topInset + 32.dp + CoverDownShift
            coverSize = maxOf(160.dp, minOf(maxWidth - ScreenSideMargin * 2f, maxHeight - coverTop - bottomInset - BelowCoverBlock))
            coverLeft = (maxWidth - coverSize) / 2f
        }
        val coverTopPx = with(density) { coverTop.toPx() }
        val bigX = with(density) { coverLeft.toPx() }
        val belowCoverStart = coverTop + coverSize + 20.dp
        val panelStart = coverLeft + coverSize + 32.dp

        val miniCornerPx = with(density) { MiniCorner.toPx() }
        val shadowPx = with(density) { 3.dp.toPx() }

        val miniColor = MaterialTheme.colorScheme.secondaryContainer
        val fullColor = MaterialTheme.colorScheme.surface

        // Запоминаем в момент расстановки, стоит ли плеер на измеренном месте: до этого он невидим
        val placedOnAnchor = remember { mutableStateOf(false) }
        val miniVisible by remember { derivedStateOf { progress.floatValue < 0.999f } }
        val fullVisible by remember { derivedStateOf { progress.floatValue > 0.001f } }

        val dragState = rememberDraggableState { delta ->
            val travel = miniRect().top.coerceAtLeast(1f)
            progress.floatValue = (progress.floatValue - delta / travel).coerceIn(0f, 1f)
        }

        Box(
            modifier = Modifier
                // позиция: из мини-плеера в левый верхний угол экрана
                .offset {
                    val m = miniRect()
                    val p = progress.floatValue
                    IntOffset(lerp(m.left, 0f, p).roundToInt(), lerp(m.top, 0f, p).roundToInt())
                }
                // размер: из мини-плеера на весь экран
                .layout { measurable, _ ->
                    placedOnAnchor.value = anchor.bounds != null
                    val m = miniRect()
                    val p = progress.floatValue
                    val w = lerp(m.width, fullW, p).roundToInt().coerceAtLeast(0)
                    val h = lerp(m.height, fullH, p).roundToInt().coerceAtLeast(0)
                    val placeable = measurable.measure(Constraints.fixed(w, h))
                    layout(w, h) { placeable.place(0, 0) }
                }
                // скругления и тень тоже «стекают» к полному экрану
                .graphicsLayer {
                    val p = progress.floatValue
                    translationX = miniOffsetX
                    // Пока место под мини-плеер не измерено, не рисуем его: иначе он мелькает поверх нижней панели
                    alpha = if (!placedOnAnchor.value && p < 0.01f) 0f else 1f
                    shape = RoundedCornerShape(CornerSize(lerp(miniCornerPx, 0f, p)))
                    clip = true
                    shadowElevation = lerp(shadowPx, 0f, p)
                }
                // цвет: secondaryContainer -> surface (только палитра Material You, без блюра)
                .drawBehind { drawRect(lerpColor(miniColor, fullColor, progress.floatValue)) }
                // Вертикальный жест и в свёрнутом, и в развёрнутом виде:
                // вниз сворачивает, вверх разворачивает вслед за пальцем
                .then(
                    Modifier.draggable(
                            state = dragState,
                            orientation = Orientation.Vertical,
                            onDragStarted = {
                                animJob?.cancel()
                                settling[0] = -1f
                            },
                            onDragStopped = { velocity ->
                                val p = progress.floatValue
                                val travel = miniRect().top.coerceAtLeast(1f)
                                val target = when {
                                    velocity > 1500f -> 0f
                                    velocity < -1500f -> 1f
                                    p < 0.5f -> 0f
                                    else -> 1f
                                }
                                settle(target, -velocity / travel)
                                onChange(target == 1f)
                            },
                        )
                )
                .then(
                    if (expanded) {
                        Modifier
                    } else {
                        // Свёрнут: свайп в сторону закрывает, тап разворачивает
                        Modifier
                            .pointerInput(Unit) {
                                detectHorizontalDragGestures(
                                    onDragStart = { dismissJob?.cancel() },
                                    onDragEnd = {
                                        val m = miniRect()
                                        if (abs(miniOffsetX) > m.width * 0.35f) {
                                            animateMiniX(if (miniOffsetX > 0f) fullW else -fullW) {
                                                audio.stop()
                                                miniOffsetX = 0f
                                            }
                                        } else {
                                            animateMiniX(0f)
                                        }
                                    },
                                    onDragCancel = { animateMiniX(0f) },
                                    onHorizontalDrag = { _, dx -> miniOffsetX += dx },
                                )
                            }
                            .clickable { onChange(true) }
                    }
                )
        ) {
            if (miniVisible) {
                MiniContent(audio = audio, progress = { progress.floatValue }, width = miniWidthDp)
            }
            if (fullVisible) {
                FullContent(
                    audio = audio,
                    likes = likes,
                    onNeedAuth = onNeedAuth,
                    expandProgress = { progress.floatValue },
                    width = fullWidthDp,
                    height = fullHeightDp,
                    topInset = topInset,
                    bottomInset = bottomInset,
                    belowCoverStart = belowCoverStart,
                    landscape = landscape,
                    panelStart = panelStart,
                    rightInset = rightInset,
                )
            }
            // Обложка — общий элемент: одна и та же картинка в мини и в полном плеере
            SharedCover(
                audio = audio,
                expanded = expanded,
                progress = { progress.floatValue },
                coverSize = coverSize,
                bigX = bigX,
                bigY = coverTopPx,
            )
        }
    }
}

// ---------------------------------------------------------------------------------------------
// Мини-плеер (содержимое). Обложки здесь нет — вместо неё пустое место, обложку рисует SharedCover.
// ---------------------------------------------------------------------------------------------

@Composable
private fun MiniContent(audio: AudioBackend, progress: () -> Float, width: Dp) {
    val state by audio.state.collectAsState()
    val title = state.title ?: return
    val showPause = state.isPlaying || state.isBuffering

    CompositionLocalProvider(LocalContentColor provides MaterialTheme.colorScheme.onSecondaryContainer) {
        Row(
            Modifier
                .wrapContentSize(Alignment.TopStart, unbounded = true)
                .size(width = width, height = MiniHeight)
                // мини-контент быстро растворяется в начале разворачивания
                .graphicsLayer { alpha = (1f - progress() / 0.25f).coerceIn(0f, 1f) }
                .padding(MiniInset),
            verticalAlignment = Alignment.CenterVertically,
        ) {
            Spacer(Modifier.size(MiniCoverSize))
            Spacer(Modifier.width(12.dp))
            Column(Modifier.weight(1f)) {
                Text(
                    title,
                    style = MaterialTheme.typography.titleMedium,
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis,
                )
                state.artist?.let {
                    Text(
                        it,
                        style = MaterialTheme.typography.bodyMedium,
                        maxLines = 1,
                        overflow = TextOverflow.Ellipsis,
                    )
                }
            }
            IconButton(
                onClick = { if (showPause) audio.pause() else audio.resume() },
                modifier = Modifier
                    .size(40.dp)
                    .clip(CircleShape)
                    .background(MaterialTheme.colorScheme.primary)
            ) {
                if (rememberDelayedFlag(state.isBuffering && !state.isPlaying, SPINNER_DELAY_MS, state.mediaId)) {
                    CircularProgressIndicator(
                        Modifier.size(22.dp),
                        strokeWidth = 2.dp,
                        color = MaterialTheme.colorScheme.onPrimary,
                    )
                } else {
                    PlayPauseIcon(showPause, Modifier.size(22.dp), MaterialTheme.colorScheme.onPrimary)
                }
            }
            Spacer(Modifier.width(8.dp))
            IconButton(onClick = { audio.stop() }, modifier = Modifier.size(40.dp)) {
                CloseIcon(Modifier.size(20.dp), MaterialTheme.colorScheme.onSurfaceVariant)
            }
        }
    }
}

// ---------------------------------------------------------------------------------------------
// Общая обложка: морфинг 48dp -> большая + наклон после удержания 1.5 с
// ---------------------------------------------------------------------------------------------

@Composable
private fun SharedCover(
    audio: AudioBackend,
    expanded: Boolean,
    progress: () -> Float,
    coverSize: Dp,
    bigX: Float,
    bigY: Float,
) {
    val density = LocalDensity.current
    val haptic = LocalHapticFeedback.current

    val playerState = audio.state.collectAsState()
    val artworkUrl by remember { derivedStateOf { playerState.value.artworkUrl } }

    val sizePx = with(density) { coverSize.toPx() }
    val miniSizePx = with(density) { MiniCoverSize.toPx() }
    val miniXY = with(density) { MiniInset.toPx() }
    val miniCornerPx = with(density) { MiniCoverCorner.toPx() }
    val fullCornerPx = with(density) { FullCoverCorner.toPx() }

    // --- Наклон обложки ---
    var tilting by remember { mutableStateOf(false) }
    var targetX by remember { mutableFloatStateOf(0f) }
    var targetY by remember { mutableFloatStateOf(0f) }
    val followSpec = remember { spring<Float>(Spring.DampingRatioNoBouncy, Spring.StiffnessMedium) }
    val releaseSpec = remember { spring<Float>(dampingRatio = 0.5f, stiffness = 220f) }
    val tiltX = animateFloatAsState(targetX, animationSpec = if (tilting) followSpec else releaseSpec, label = "tiltX")
    val tiltY = animateFloatAsState(targetY, animationSpec = if (tilting) followSpec else releaseSpec, label = "tiltY")
    val press = animateFloatAsState(
        if (tilting) 1f else 0f,
        animationSpec = if (tilting) followSpec else releaseSpec,
        label = "press",
    )

    LaunchedEffect(expanded) {
        if (!expanded) {
            tilting = false
            targetX = 0f
            targetY = 0f
        }
    }

    Box(
        Modifier
            .wrapContentSize(Alignment.TopStart, unbounded = true)
            .size(coverSize)
            // морфинг: позиция + масштаб от левого верхнего угла
            .graphicsLayer {
                val p = progress()
                val s = lerp(miniSizePx / sizePx, 1f, p)
                transformOrigin = TransformOrigin(0f, 0f)
                scaleX = s
                scaleY = s
                translationX = lerp(miniXY, bigX, p)
                translationY = lerp(miniXY, bigY, p)
            }
    ) {
        Box(
            Modifier
                .fillMaxSize()
                .then(
                    if (expanded) {
                        // Жест стоит до graphicsLayer, чтобы координаты не искажались поворотом.
                        // Не потребляем события, пока не прошло 1.5 с, поэтому свайп вниз работает как обычно.
                        Modifier.pointerInput(Unit) {
                            awaitEachGesture {
                                val down = awaitFirstDown(requireUnconsumed = false)
                                // Ждём 1.5 с: если палец поднят или сдвинут раньше — это не наш жест
                                val interrupted = withTimeoutOrNull(HOLD_TO_TILT_MS) {
                                    while (true) {
                                        val event = awaitPointerEvent()
                                        val change = event.changes.firstOrNull { it.id == down.id }
                                        if (change == null || !change.pressed ||
                                            (change.position - down.position).getDistance() > viewConfiguration.touchSlop
                                        ) {
                                            return@withTimeoutOrNull
                                        }
                                    }
                                }
                                if (interrupted != null) return@awaitEachGesture

                                // Прошло 1.5 с: включаем наклон
                                haptic.performHapticFeedback(HapticFeedbackType.LongPress)
                                tilting = true

                                fun updateTarget(pos: Offset) {
                                    targetX = (pos.x / size.width * 2f - 1f).coerceIn(-1f, 1f)
                                    targetY = (pos.y / size.height * 2f - 1f).coerceIn(-1f, 1f)
                                }
                                updateTarget(down.position)

                                while (true) {
                                    val event = awaitPointerEvent()
                                    val change = event.changes.firstOrNull { it.id == down.id } ?: break
                                    change.consume()
                                    if (!change.pressed) break
                                    updateTarget(change.position)
                                }

                                tilting = false
                                targetX = 0f
                                targetY = 0f
                            }
                        }
                    } else {
                        Modifier
                    }
                )
                .graphicsLayer {
                    rotationY = tiltX.value * MAX_TILT_DEG
                    rotationX = -tiltY.value * MAX_TILT_DEG
                    cameraDistance = 14f * this.density
                    val k = 1f - 0.04f * press.value
                    scaleX = k
                    scaleY = k
                }
        ) {
            Box(
                Modifier
                    .fillMaxSize()
                    // скругление меняется вместе с морфингом (с поправкой на масштаб)
                    .graphicsLayer {
                        val p = progress()
                        val s = lerp(miniSizePx / sizePx, 1f, p)
                        val r = lerp(miniCornerPx, fullCornerPx, p) / s
                        shape = RoundedCornerShape(CornerSize(r))
                        clip = true
                    }
            ) {
                CoverImage(artworkUrl, Modifier.fillMaxSize(), RoundedCornerShape(0.dp))

                // Блик уходит на сторону, противоположную пальцу
                Canvas(Modifier.fillMaxSize()) {
                    val pr = press.value
                    if (pr > 0f) {
                        val center = Offset(
                            size.width * (1f - tiltX.value) / 2f,
                            size.height * (1f - tiltY.value) / 2f,
                        )
                        drawRect(
                            Brush.radialGradient(
                                colors = listOf(Color.White.copy(alpha = 0.32f * pr), Color.Transparent),
                                center = center,
                                radius = size.minDimension * 0.75f,
                            )
                        )
                    }
                }

                // Значок источника текущего трека (Jamendo, Audius, Deezer), виден только в полном плеере
                ServiceBadge(
                    playerState.value.provider?.toService(),
                    Modifier
                        .align(Alignment.BottomEnd)
                        .padding(16.dp)
                        .graphicsLayer { alpha = ((progress() - 0.6f) / 0.4f).coerceIn(0f, 1f) },
                    36.dp,
                )
            }
        }
    }
}

// ---------------------------------------------------------------------------------------------
// Полный плеер (содержимое без обложки и без блюра)
// ---------------------------------------------------------------------------------------------

@Composable
private fun FullContent(
    audio: AudioBackend,
    likes: LikesRepository,
    onNeedAuth: () -> Unit,
    expandProgress: () -> Float,
    width: Dp,
    height: Dp,
    topInset: Dp,
    bottomInset: Dp,
    belowCoverStart: Dp,
    landscape: Boolean,
    panelStart: Dp,
    rightInset: Dp,
) {
    val state by audio.state.collectAsState()
    var dragValue by remember { mutableStateOf<Float?>(null) }
    val likedTracks by likes.tracks.collectAsState()
    val signedIn by likes.signedIn.collectAsState()
    val likeError by likes.error.collectAsState()
    val currentTrack = state.asTrack()
    val liked = likedTracks.containsTrack(currentTrack)
    var showDetails by remember { mutableStateOf(false) }

    val trackProgress = if (state.durationMs > 0) {
        (state.positionMs.toFloat() / state.durationMs).coerceIn(0f, 1f)
    } else {
        0f
    }
    val shownProgress = dragValue ?: trackProgress
    val shownPositionMs = dragValue?.let { (it * state.durationMs).toLong() } ?: state.positionMs
    val showPause = state.isPlaying || state.isBuffering
    val active = MaterialTheme.colorScheme.primary
    val inactive = MaterialTheme.colorScheme.onSurfaceVariant

    CompositionLocalProvider(LocalContentColor provides MaterialTheme.colorScheme.onSurface) {
        Box(
            Modifier
                .wrapContentSize(Alignment.TopStart, unbounded = true)
                .size(width = width, height = height)
                // полный контент проявляется во второй половине разворачивания
                .graphicsLayer { alpha = ((expandProgress() - 0.3f) / 0.5f).coerceIn(0f, 1f) }
        ) {
            // Ручка для свайпа
            Box(
                Modifier
                    .align(Alignment.TopCenter)
                    .padding(top = topInset + 8.dp)
                    .width(40.dp)
                    .height(4.dp)
                    .clip(CircleShape)
                    .background(MaterialTheme.colorScheme.onSurfaceVariant.copy(alpha = 0.3f))
            )

            Column(
                modifier = if (landscape) {
                    // Альбом: обложка слева, всё остальное в правой колонке по центру высоты
                    Modifier
                        .fillMaxSize()
                        .padding(
                            start = panelStart,
                            end = rightInset + 32.dp,
                            top = topInset + 16.dp,
                            bottom = bottomInset + 8.dp,
                        )
                } else {
                    Modifier
                        .fillMaxSize()
                        .padding(horizontal = ScreenSideMargin)
                        .padding(bottom = bottomInset)
                },
                verticalArrangement = if (landscape) Arrangement.Center else Arrangement.Top,
            ) {
                // Здесь сверху лежит общая обложка (её рисует SharedCover)
                if (!landscape) Spacer(Modifier.height(belowCoverStart))

                Text(
                    state.title ?: "",
                    style = if (landscape) MaterialTheme.typography.titleLarge else MaterialTheme.typography.headlineSmall,
                    maxLines = if (landscape) 1 else 2,
                    overflow = TextOverflow.Ellipsis,
                    modifier = Modifier.fillMaxWidth(),
                )
                Text(
                    state.artist ?: "",
                    style = MaterialTheme.typography.bodyLarge,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis,
                    modifier = Modifier.fillMaxWidth(),
                )

                (state.error ?: likeError)?.let {
                    Text(it, color = MaterialTheme.colorScheme.error, style = MaterialTheme.typography.bodySmall)
                }

                Row(
                    Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically,
                ) {
                    IconButton(onClick = { showDetails = true }) {
                        MoreIcon()
                    }
                    Row {
                        IconButton(onClick = {
                            when {
                                currentTrack == null -> Unit
                                !signedIn -> onNeedAuth()
                                else -> likes.toggle(currentTrack)
                            }
                        }) {
                            HeartIcon(liked, tint = if (liked) active else inactive)
                        }
                        IconButton(onClick = { audio.cycleRepeat() }) {
                            RepeatIcon(
                                one = state.repeatMode == RepeatMode.ONE,
                                tint = if (state.repeatMode == RepeatMode.OFF) inactive else active,
                            )
                        }
                    }
                }

                // Волнистый таймлайн; поверх невидимый Slider принимает перемотку
                Box(Modifier.fillMaxWidth().height(40.dp), contentAlignment = Alignment.Center) {
                    LinearWavyProgressIndicator(
                        progress = { shownProgress },
                        amplitude = { if (state.isPlaying) 1f else 0f },
                        modifier = Modifier.fillMaxWidth().padding(horizontal = 12.dp),
                    )
                    Slider(
                        value = shownProgress,
                        onValueChange = { dragValue = it },
                        onValueChangeFinished = {
                            dragValue?.let { audio.seekTo((it * state.durationMs).toLong()) }
                            dragValue = null
                        },
                        enabled = state.durationMs > 0,
                        modifier = Modifier.fillMaxWidth().alpha(0f),
                    )
                }
                Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.SpaceBetween) {
                    Text(formatTime(shownPositionMs), style = MaterialTheme.typography.labelMedium)
                    Text(formatTime(state.durationMs), style = MaterialTheme.typography.labelMedium)
                }

                if (landscape) Spacer(Modifier.height(8.dp)) else Spacer(Modifier.weight(1f))

                Row(
                    if (landscape) {
                        Modifier.fillMaxWidth().padding(top = 8.dp)
                    } else {
                        Modifier.fillMaxWidth().padding(top = 16.dp, bottom = 80.dp)
                    },
                    horizontalArrangement = Arrangement.SpaceEvenly,
                    verticalAlignment = Alignment.CenterVertically,
                ) {
                    IconButton(onClick = { audio.previous() }, enabled = state.hasPrevious, modifier = Modifier.size(60.dp)) {
                        SkipIcon(next = false, modifier = Modifier.size(36.dp))
                    }
                    FilledIconButton(
                        onClick = { if (showPause) audio.pause() else audio.resume() },
                        modifier = Modifier.size(86.dp),
                    ) {
                        if (rememberDelayedFlag(state.isBuffering && !state.isPlaying, SPINNER_DELAY_MS, state.mediaId)) {
                            CircularProgressIndicator(
                                Modifier.size(40.dp),
                                color = LocalContentColor.current,
                                strokeWidth = 3.dp,
                            )
                        } else {
                            PlayPauseIcon(showPause, Modifier.size(48.dp))
                        }
                    }
                    IconButton(onClick = { audio.next() }, enabled = state.hasNext, modifier = Modifier.size(60.dp)) {
                        SkipIcon(next = true, modifier = Modifier.size(36.dp))
                    }
                }
            }
        }
    }

    if (showDetails) {
        AlertDialog(
            onDismissRequest = { showDetails = false },
            confirmButton = { TextButton(onClick = { showDetails = false }) { Text("Закрыть") } },
            title = { Text(state.title ?: "") },
            text = { Text("Исполнитель: ${state.artist ?: "—"}\nИсточник: ${state.provider?.toService()?.title ?: "—"}") },
        )
    }
}

private fun formatTime(ms: Long): String {
    val totalSeconds = ms / 1000
    return "${totalSeconds / 60}:${(totalSeconds % 60).toString().padStart(2, '0')}"

 * Мини-плеер и полный плеер Spotify: тот же морфинг, что у ExpandablePlayer,
 * но отдельный компонент: ничего общего с AudioBackend и нашими источниками.
 * Ставить в корневой Box поверх основного контента.
 */
@Composable
fun SpotifyExpandablePlayer(remote: SpotifyRemote, audio: AudioBackend, modifier: Modifier = Modifier) {
    val ui = LocalSpotifyMini.current
    val anchor = ui.anchor
    val ownAnchor = LocalMiniPlayerAnchor.current
    val stateHolder = remote.state.collectAsState()
    val audioHolder = audio.state.collectAsState()
    // Сырое условие. Spotify при смене трека может на секунду прислать «пустое» состояние
    val rawHasTrack by remember {
        derivedStateOf {
            val st = stateHolder.value
            st.connected && st.premium != false && st.title != null &&
                audioHolder.value.title == null &&
                !(ui.dismissedUri != null && ui.dismissedUri == st.uri && st.isPaused)
        }
    }
    val audioBusy by remember { derivedStateOf { audioHolder.value.title != null } }
    // Последнее состояние с названием трека: его показываем, пока длится короткий разрыв
    val held = remember { mutableStateOf(stateHolder.value) }
    LaunchedEffect(Unit) {
        snapshotFlow { stateHolder.value }.collect { if (it.title != null) held.value = it }
    }
    // Spotify на миг присылает пустое состояние при смене трека: держим плеер до 1,2 с, но только в неявных случаях
    var everShown by remember { mutableStateOf(false) }
    var graceExpired by remember { mutableStateOf(false) }
    val explicitHide by remember {
        derivedStateOf {
            val st = stateHolder.value
            // Наш плеер должен сначала занять своё место, и только потом Spotify скрывается: без разрыва между ними
            (audioBusy && ownAnchor.bounds != null) || !st.connected || st.premium == false ||
                (ui.dismissedUri != null && ui.dismissedUri == st.uri && st.isPaused)
        }
    }
    LaunchedEffect(rawHasTrack) {
        if (rawHasTrack) {
            everShown = true
            graceExpired = false
        } else {
            delay(1200)
            graceExpired = true
        }
    }
    // Считаем прямо в композиции: плеер скрывается и появляется в том же кадре, что и смена источника
    val hasTrack = rawHasTrack || (everShown && !graceExpired && !explicitHide)
    val expanded = ui.expanded
    val scope = rememberCoroutineScope()

    SideEffect { ui.shown = hasTrack }
    LaunchedEffect(Unit) {
        snapshotFlow { Triple(rawHasTrack, audioHolder.value.title != null, stateHolder.value.title) }.collect {
            println("Solara: timeline spotifyMini=${it.first} audioTitle=${it.second} spTitle=${it.third} spPaused=${stateHolder.value.isPaused} audioPlaying=${audioHolder.value.isPlaying}")
        }
    }
    DisposableEffect(Unit) { onDispose { ui.shown = false } }
    // Снова нажали play: закрытый ранее трек опять показываем
    LaunchedEffect(Unit) {
        snapshotFlow { stateHolder.value.isPaused }.collect { paused -> if (!paused) ui.dismissedUri = null }
    }

    val progress = remember { mutableFloatStateOf(if (expanded) 1f else 0f) }
    var animJob by remember { mutableStateOf<Job?>(null) }
    val settling = remember { floatArrayOf(-1f) }

    fun settle(target: Float, velocity: Float = 0f) {
        settling[0] = target
        animJob?.cancel()
        animJob = scope.launch {
            animate(
                initialValue = progress.floatValue,
                targetValue = target,
                initialVelocity = velocity,
                animationSpec = spring(
                    dampingRatio = Spring.DampingRatioNoBouncy,
                    stiffness = Spring.StiffnessMediumLow,
                ),
            ) { v, _ -> progress.floatValue = v.coerceIn(0f, 1f) }
            settling[0] = -1f
        }
    }

    LaunchedEffect(expanded) {
        val target = if (expanded) 1f else 0f
        if (settling[0] != target && progress.floatValue != target) settle(target)
    }

    LaunchedEffect(hasTrack) {
        if (!hasTrack) {
            animJob?.cancel()
            settling[0] = -1f
            progress.floatValue = 0f
            ui.expanded = false
        }
    }
    if (!hasTrack) return

    PlatformBackHandler(enabled = expanded, onBack = { ui.expanded = false })

    var origin by remember { mutableStateOf(Offset.Zero) }
    var miniOffsetX by remember { mutableFloatStateOf(0f) }
    var dismissJob by remember { mutableStateOf<Job?>(null) }

    fun animateMiniX(to: Float, after: () -> Unit = {}) {
        dismissJob?.cancel()
        dismissJob = scope.launch {
            animate(initialValue = miniOffsetX, targetValue = to) { v, _ -> miniOffsetX = v }
            after()
        }
    }

    fun dismissMini() {
        ui.dismissedUri = stateHolder.value.uri
        remote.pause()
    }

    BoxWithConstraints(
        modifier
            .fillMaxSize()
            .onGloballyPositioned { origin = it.positionInRoot() }
    ) {
        val density = LocalDensity.current
        val fullW = constraints.maxWidth.toFloat()
        val fullH = constraints.maxHeight.toFloat()
        val fullWidthDp = maxWidth
        val fullHeightDp = maxHeight

        val topInset = WindowInsets.statusBars.asPaddingValues().calculateTopPadding()
        val bottomInset = WindowInsets.navigationBars.asPaddingValues().calculateBottomPadding()

        val fbMargin = with(density) { 8.dp.toPx() }
        val fbBottom = with(density) { (bottomInset + 8.dp).toPx() }
        val miniHpx = with(density) { MiniHeight.toPx() }
        fun miniRect(): Rect {
            val b = anchor.bounds
            return if (b != null) {
                Rect(b.left - origin.x, b.top - origin.y, b.right - origin.x, b.bottom - origin.y)
            } else {
                Rect(fbMargin, fullH - fbBottom - miniHpx, fullW - fbMargin, fullH - fbBottom)
            }
        }
        val miniNow = miniRect()
        val miniWidthDp = with(density) { miniNow.width.toDp() }

        val layoutDirection = LocalLayoutDirection.current
        val sideInsets = WindowInsets.systemBars.union(WindowInsets.displayCutout)
        val leftInset = with(density) { sideInsets.getLeft(density, layoutDirection).toDp() }
        val rightInset = with(density) { sideInsets.getRight(density, layoutDirection).toDp() }
        val landscape = fullW > fullH

        val coverTop: Dp
        val coverSize: Dp
        val coverLeft: Dp
        if (landscape) {
            val availH = maxHeight - topInset - bottomInset
            coverSize = maxOf(120.dp, minOf(availH - 24.dp, maxWidth * 0.42f))
            coverTop = topInset + (availH - coverSize) / 2f
            coverLeft = leftInset + 32.dp
        } else {
            coverTop = topInset + 32.dp + SpCoverDownShift
            coverSize = maxOf(160.dp, minOf(maxWidth - SpScreenSideMargin * 2f, maxHeight - coverTop - bottomInset - SpBelowCoverBlock))
            coverLeft = (maxWidth - coverSize) / 2f
        }
        val coverTopPx = with(density) { coverTop.toPx() }
        val bigX = with(density) { coverLeft.toPx() }
        val belowCoverStart = coverTop + coverSize + 20.dp
        val panelStart = coverLeft + coverSize + 32.dp

        val miniCornerPx = with(density) { SpMiniCorner.toPx() }
        val shadowPx = with(density) { 3.dp.toPx() }
        val miniColor = MaterialTheme.colorScheme.secondaryContainer
        val fullColor = MaterialTheme.colorScheme.surface

        // Запоминаем в момент расстановки, стоит ли плеер на измеренном месте: до этого он невидим
        val placedOnAnchor = remember { mutableStateOf(false) }
        val miniVisible by remember { derivedStateOf { progress.floatValue < 0.999f } }
        val fullVisible by remember { derivedStateOf { progress.floatValue > 0.001f } }

        val dragState = rememberDraggableState { delta ->
            val travel = miniRect().top.coerceAtLeast(1f)
            progress.floatValue = (progress.floatValue - delta / travel).coerceIn(0f, 1f)
        }

        Box(
            modifier = Modifier
                .offset {
                    val m = miniRect()
                    val p = progress.floatValue
                    IntOffset(lerp(m.left, 0f, p).roundToInt(), lerp(m.top, 0f, p).roundToInt())
                }
                .layout { measurable, _ ->
                    placedOnAnchor.value = anchor.bounds != null
                    val m = miniRect()
                    val p = progress.floatValue
                    val w = lerp(m.width, fullW, p).roundToInt().coerceAtLeast(0)
                    val h = lerp(m.height, fullH, p).roundToInt().coerceAtLeast(0)
                    val placeable = measurable.measure(Constraints.fixed(w, h))
                    layout(w, h) { placeable.place(0, 0) }
                }
                .graphicsLayer {
                    val p = progress.floatValue
                    translationX = miniOffsetX
                    // Пока место под мини-плеер не измерено, не рисуем его: иначе он мелькает поверх нижней панели
                    alpha = if (!placedOnAnchor.value && p < 0.01f) 0f else 1f
                    shape = RoundedCornerShape(CornerSize(lerp(miniCornerPx, 0f, p)))
                    clip = true
                    shadowElevation = lerp(shadowPx, 0f, p)
                }
                .drawBehind { drawRect(lerpColor(miniColor, fullColor, progress.floatValue)) }
                .draggable(
                    state = dragState,
                    orientation = Orientation.Vertical,
                    onDragStarted = {
                        animJob?.cancel()
                        settling[0] = -1f
                    },
                    onDragStopped = { velocity ->
                        val p = progress.floatValue
                        val travel = miniRect().top.coerceAtLeast(1f)
                        val target = when {
                            velocity > 1500f -> 0f
                            velocity < -1500f -> 1f
                            p < 0.5f -> 0f
                            else -> 1f
                        }
                        settle(target, -velocity / travel)
                        ui.expanded = target == 1f
                    },
                )
                .then(
                    if (expanded) {
                        Modifier
                    } else {
                        Modifier
                            .pointerInput(Unit) {
                                detectHorizontalDragGestures(
                                    onDragStart = { dismissJob?.cancel() },
                                    onDragEnd = {
                                        val m = miniRect()
                                        if (abs(miniOffsetX) > m.width * 0.35f) {
                                            animateMiniX(if (miniOffsetX > 0f) fullW else -fullW) {
                                                dismissMini()
                                                miniOffsetX = 0f
                                            }
                                        } else {
                                            animateMiniX(0f)
                                        }
                                    },
                                    onDragCancel = { animateMiniX(0f) },
                                    onHorizontalDrag = { _, dx -> miniOffsetX += dx },
                                )
                            }
                            .clickable { ui.expanded = true }
                    }
                )
        ) {
            if (miniVisible) {
                SpMiniContent(remote, held, { progress.floatValue }, miniWidthDp) { dismissMini() }
            }
            if (fullVisible) {
                SpFullContent(
                    remote = remote,
                    stateHolder = held,
                    expandProgress = { progress.floatValue },
                    width = fullWidthDp,
                    height = fullHeightDp,
                    topInset = topInset,
                    bottomInset = bottomInset,
                    belowCoverStart = belowCoverStart,
                    landscape = landscape,
                    panelStart = panelStart,
                    rightInset = rightInset,
                )
            }
            SpSharedCover(
                stateHolder = held,
                progress = { progress.floatValue },
                coverSize = coverSize,
                bigX = bigX,
                bigY = coverTopPx,
            )
        }
    }
}

/** Общая обложка: морфинг 48 dp -> большая. Без наклона, блика, значков и блюра, углы 4 dp, без обрезки. */
@Composable
private fun SpSharedCover(
    stateHolder: State<SpotifyState>,
    progress: () -> Float,
    coverSize: Dp,
    bigX: Float,
    bigY: Float,
) {
    val density = LocalDensity.current
    val cover by remember { derivedStateOf { stateHolder.value.cover } }

    val sizePx = with(density) { coverSize.toPx() }
    val miniSizePx = with(density) { SpMiniCoverSize.toPx() }
    val miniXY = with(density) { SpMiniInset.toPx() }
    val cornerPx = with(density) { SpCoverCorner.toPx() }

    Box(
        Modifier
            .wrapContentSize(Alignment.TopStart, unbounded = true)
            .size(coverSize)
            .graphicsLayer {
                val p = progress()
                val sc = lerp(miniSizePx / sizePx, 1f, p)
                transformOrigin = TransformOrigin(0f, 0f)
                scaleX = sc
                scaleY = sc
                translationX = lerp(miniXY, bigX, p)
                translationY = lerp(miniXY, bigY, p)
            }
    ) {
        Box(
            Modifier
                .fillMaxSize()
                .graphicsLayer {
                    // 4 dp на экране при любом масштабе
                    val sc = lerp(miniSizePx / sizePx, 1f, progress())
                    shape = RoundedCornerShape(CornerSize(cornerPx / sc))
                    clip = true
                }
                .background(MaterialTheme.colorScheme.surfaceVariant)
        ) {
            cover?.let { Image(it, null, Modifier.fillMaxSize(), contentScale = ContentScale.Fit) }
        }
    }
}

            onDismissRequest = { showDetails = false },
            confirmButton = { TextButton(onClick = { showDetails = false }) { Text("Закрыть") } },
            title = { Text(s.title ?: "") },
            text = { Text("Исполнитель: ${s.artist ?: "—"}\nИсточник: Spotify") },
        )
    }
}

private fun spTime(ms: Long): String {
    val total = ms / 1000
    return "${total / 60}:${(total % 60).toString().padStart(2, '0')}"

// Сколько показываем вращение заставки, когда приложение готово: минимум и потолок (например, нет сети).
// Время считается уже после входа логотипа и первой сборки интерфейса.
private const val SPLASH_MIN_MS = 450L
private const val SPLASH_MAX_MS = 1000L

/** Всё тяжёлое, что создаётся в фоне, пока главный поток рисует заставку. */
private class AppBoot(
    val providers: List<MusicProvider>,
    val authRepo: AuthRepository,
)

@Composable
fun App() {
    SolaraTheme {
        Surface(Modifier.fillMaxSize()) {
            // HTTP-клиенты, Supabase и провайдеры создаём в фоновом потоке: главный в это время свободен для анимации
            val boot by produceState<AppBoot?>(initialValue = null) {
                value = withContext(Dispatchers.Default) {
                    AppBoot(
                        providers = listOf<MusicProvider>(
                            JamendoProvider(Secrets.JAMENDO_CLIENT_ID),
                            AudiusProvider(Secrets.AUDIUS_API_KEY),
                            DeezerProvider(),
                        ),
                        authRepo = AuthRepository(),
                    )
                }
            }
            var entered by remember { mutableStateOf(false) }
            var spinning by remember { mutableStateOf(false) }
            var leaving by remember { mutableStateOf(false) }
            var splashDone by remember { mutableStateOf(false) }

            // Тяжёлую первую сборку интерфейса запускаем, когда готов фон и логотип уже вошёл:
            // подвисание (если оно будет) приходится на неподвижную заставку, а не на вращение.
            val ready = if (entered) boot else null

            Box(Modifier.fillMaxSize()) {
                if (ready != null) {
                    AppContent(ready)

                    val authState by ready.authRepo.state.collectAsState()
                    // Вход «готов», когда понятно, гость это или вошедший пользователь с загруженным профилем
                    val authSettled = when (val s = authState) {
                        AuthState.Loading -> false
                        is AuthState.SignedIn -> s.profile != null
                        else -> true
                    }
                    val settled by rememberUpdatedState(authSettled)
                    LaunchedEffect(Unit) {
                        // два кадра: тяжёлый первый кадр интерфейса уже нарисован, вращение стартует с чистого листа
                        withFrameNanos { }
                        withFrameNanos { }
                        spinning = true
                        delay(SPLASH_MIN_MS)
                        withTimeoutOrNull(SPLASH_MAX_MS - SPLASH_MIN_MS) {
                            snapshotFlow { settled }.first { it }
                        }
                        leaving = true
                    }
                }
                if (!splashDone) {
                    SplashScreen(
                        spinning = spinning,
                        leaving = leaving,
                        onEntered = { entered = true },
                        onFinished = { splashDone = true },
                    )
                }
            }
        }
    }
}

@Composable
private fun AppContent(boot: AppBoot) {
    val providers = boot.providers
    val authRepo = boot.authRepo
    val audio = rememberAudioBackend()
    val likes = remember { LikesRepository(authRepo) }
    val authState by authRepo.state.collectAsState()
    val pending by authRepo.pending.collectAsState()
    var showPlayer by remember { mutableStateOf(false) }
    var showAuth by remember { mutableStateOf(false) }
    val miniAnchor = remember { MiniPlayerAnchor() }
    val spotify = LocalSpotifyRemote.current
    val spotifyMini = remember { SpotifyMiniState() }
    val focusManager = LocalFocusManager.current
    // Снимаем фокус с поля поиска, иначе его «капля» остаётся поверх плеера
    LaunchedEffect(showPlayer) { if (showPlayer) focusManager.clearFocus() }
    // Вошли (в том числе через Google после возврата из браузера): закрываем экран входа
    LaunchedEffect(authState) {
        if (authState is AuthState.SignedIn) showAuth = false
    }
    CompositionLocalProvider(
        LocalMiniPlayerAnchor provides miniAnchor,
        LocalSpotifyMini provides spotifyMini,
    ) {
        Box(Modifier.fillMaxSize()) {
            MainScreen(
                providers = providers,
                audio = audio,
                authRepo = authRepo,
                likes = likes,
                onOpenPlayer = { showPlayer = true },
                onOpenAuth = { showAuth = true },
            )
            // Мини-плеер и полный плеер — один компонент: плавно перетекает из одного в другой
            ExpandablePlayer(
                audio = audio,
                likes = likes,
                expanded = showPlayer,
                onExpandedChange = { showPlayer = it },
                // Гость нажал ♡: сворачиваем плеер и открываем вход
                onNeedAuth = {
                    showPlayer = false
                    showAuth = true
                },
            )
            // Мини-плеер Spotify: отдельный компонент, не связан с нашим AudioBackend-плеером
            if (spotify != null) {
                SpotifyExpandablePlayer(remote = spotify, audio = audio)
            }
            // Экран входа и регистрации поверх всего
            if (showAuth) {
                PlatformBackHandler(enabled = true, onBack = { showAuth = false })
                AuthScreen(repo = authRepo, onClose = { showAuth = false })
            }
            // После регистрации: «Ожидаем подтверждение письма».
            // Закрывается сам, когда письмо подтверждено и пользователь вошёл.
            pending?.let { p ->
                AwaitEmailScreen(
                    repo = authRepo,
                    pending = PendingSignUp(p.email, p.password),
                    onBack = { authRepo.cancelPending() },
                )
            }
            // Первый вход: ник, аватарка, сервисы, синхронизация палитры.
            // Закрывается сам, когда профиль сохранён (onboarded = true).
            val signedIn = authState as? AuthState.SignedIn
            val profile = signedIn?.profile
            if (profile != null && !profile.onboarded) {
                OnboardingScreen(repo = authRepo, profile = profile)
            }
        }
    }
}
