package com.unknowstorm.musicapp.spotify

import android.app.Activity
import android.content.Intent
import android.net.Uri
import android.os.Handler
import android.os.Looper
import androidx.compose.ui.graphics.asImageBitmap
import com.spotify.android.appremote.api.ConnectionParams
import com.spotify.android.appremote.api.Connector
import com.spotify.android.appremote.api.SpotifyAppRemote
import com.spotify.protocol.client.Subscription
import com.spotify.sdk.android.auth.AuthorizationClient
import com.spotify.sdk.android.auth.AuthorizationRequest
import com.spotify.sdk.android.auth.AuthorizationResponse
import com.spotify.protocol.types.Image
import com.spotify.protocol.types.PlayerState
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update

private const val AUTH_REQUEST = 7341

class SpotifyRemoteImpl(private val activity: Activity) : SpotifyRemote {
    companion object {
        @Volatile var current: SpotifyRemoteImpl? = null
    }

    fun pauseNow() { remote?.playerApi?.pause() }

    private val _state = MutableStateFlow(SpotifyState(appInstalled = SpotifyAppRemote.isSpotifyInstalled(activity)))
    override val state: StateFlow<SpotifyState> = _state.asStateFlow()

    private var remote: SpotifyAppRemote? = null
    private var sub: Subscription<PlayerState>? = null
    private var lastTrackUri: String? = null
    private val handler = Handler(Looper.getMainLooper())
    private val prefs = activity.getSharedPreferences("spotify_remote", 0)
    private var silent = false
    private var ignoreIdle = false
    private var optimisticKey: String? = null
    private val connectTimeout = Runnable {
        _state.update {
            if (it.connecting) it.copy(
                connecting = false,
                error = if (silent) null else "Spotify не ответил. Откройте приложение Spotify и разрешите ему всплывающие окна в фоне, затем повторите.",
            ) else it
        }
    }

    private val ticker = object : Runnable {
        override fun run() {
            _state.update { s ->
                if (s.connected && !s.isPaused && s.durationMs > 0)
                    s.copy(positionMs = (s.positionMs + 500).coerceAtMost(s.durationMs))
                else s
            }
            handler.postDelayed(this, 500)
        }
    }

    override fun connect() {
        silent = false
        ignoreIdle = false
        startConnect()
    }

    /** Тихое подключение при запуске: только если пользователь уже подключал Spotify и не отключал. */
    fun autoConnect() {
        if (!prefs.getBoolean("auto", false)) return
        if (remote?.isConnected == true || _state.value.connecting) return
        silent = true
        ignoreIdle = true
        // Разрешение уже выдано при ручном подключении: окно входа пропускаем, подключаемся напрямую
        connectRemote()
    }

    private fun startConnect() {
        if (remote?.isConnected == true) return
        if (!SpotifyAppRemote.isSpotifyInstalled(activity)) {
            _state.update { it.copy(appInstalled = false) }
            return
        }
        if (SpotifyConfig.CLIENT_ID.isBlank()) {
            _state.update { it.copy(error = "Не задан Client ID Spotify") }
            return
        }
        _state.update { it.copy(appInstalled = true, connecting = true, error = null) }
        handler.removeCallbacks(connectTimeout)
        // Окно разрешения открываем из Solara, пока она на экране: из фона Android его блокирует
        val request = AuthorizationRequest.Builder(
            SpotifyConfig.CLIENT_ID, AuthorizationResponse.Type.TOKEN, SpotifyConfig.REDIRECT_URI,
        ).setScopes(arrayOf("app-remote-control")).build()
        AuthorizationClient.openLoginActivity(activity, AUTH_REQUEST, request)
    }

    /** Вызывается из MainActivity. Токен из ответа не используется и не сохраняется. */
    fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
        if (requestCode != AUTH_REQUEST) return
        val response = AuthorizationClient.getResponse(resultCode, data)
        if (response.type == AuthorizationResponse.Type.TOKEN) {
            connectRemote()
        } else {
            _state.update {
                it.copy(connecting = false, error = "Spotify не подтвердил доступ. Проверьте, что ваш аккаунт добавлен в User Management.")
            }
        }
    }

    private fun connectRemote() {
        if (remote?.isConnected == true) return
        if (!SpotifyAppRemote.isSpotifyInstalled(activity)) {
            _state.update { it.copy(appInstalled = false) }
            return
        }
        if (SpotifyConfig.CLIENT_ID.isBlank()) {
            _state.update { it.copy(error = "Не задан Client ID Spotify") }
            return
        }
        _state.update { it.copy(appInstalled = true, connecting = true, error = null) }
        handler.removeCallbacks(connectTimeout)
        handler.postDelayed(connectTimeout, 20_000)
        val params = ConnectionParams.Builder(SpotifyConfig.CLIENT_ID)
            .setRedirectUri(SpotifyConfig.REDIRECT_URI)
            .showAuthView(!silent)
            .build()
        SpotifyAppRemote.connect(activity, params, object : Connector.ConnectionListener {
            override fun onConnected(r: SpotifyAppRemote) {
                handler.removeCallbacks(connectTimeout)
                remote = r
                prefs.edit().putBoolean("auto", true).apply()
                current = this@SpotifyRemoteImpl
                runCatching { activity.startService(Intent(activity, SpotifyTaskGuard::class.java)) }
                _state.update { it.copy(connecting = false, connected = true, error = null) }
                r.userApi.capabilities.setResultCallback { c ->
                    _state.update { it.copy(premium = c.canPlayOnDemand) }
                }
                subscribe(r)
                handler.removeCallbacks(ticker)
                handler.postDelayed(ticker, 500)
            }

            override fun onFailure(t: Throwable) {
                handler.removeCallbacks(connectTimeout)
                val text = when (t.javaClass.simpleName) {
                    "CouldNotFindSpotifyApp" -> "Приложение Spotify не найдено"
                    "NotLoggedInException" -> "Войдите в аккаунт в приложении Spotify"
                    "UserNotAuthorizedException" -> "Разрешите доступ в окне Spotify. Аккаунт должен быть добавлен в User Management."
                    "AuthenticationFailedException" -> "Spotify отклонил приложение: проверьте Client ID, SHA-1 и пакет в Dashboard"
                    else -> "Не удалось подключиться к Spotify"
                }
                _state.update { it.copy(connecting = false, connected = false, error = if (silent) null else text) }
            }
        })
    }

    private fun subscribe(r: SpotifyAppRemote) {
        sub?.cancel()
        val subscription = r.playerApi.subscribeToPlayerState()
        subscription.setEventCallback { ps ->
            // После тихого автоподключения не показываем то, что Spotify играл раньше: ждём, пока что-то реально заиграет
            if (ignoreIdle) {
                if (ps.isPaused) return@setEventCallback
                ignoreIdle = false
            }
            val track = ps.track
            _state.update {
                it.copy(
                    uri = track?.uri,
                    title = track?.name,
                    artist = track?.artist?.name,
                    isPaused = ps.isPaused,
                    positionMs = ps.playbackPosition,
                    durationMs = track?.duration ?: 0L,
                    error = null,
                )
            }
            if (track != null && track.uri != lastTrackUri) {
                lastTrackUri = track.uri
                val keepCover = optimisticKey != null && optimisticKey == track.imageUri?.raw
                optimisticKey = null
                requestCover(track.imageUri)
                _state.update { st ->
                    val item = SpotifyItem(track.uri, track.name, track.artist?.name, imageKey = track.imageUri?.raw)
                    st.copy(cover = if (keepCover) st.cover else null, saved = false,
                        history = (listOf(item) + st.history.filterNot { h -> h.uri == track.uri }).take(30))
                }
                r.imagesApi.getImage(track.imageUri, Image.Dimension.LARGE).setResultCallback { bmp ->
                    if (lastTrackUri == track.uri) _state.update { it.copy(cover = bmp.asImageBitmap()) }
                }
                r.userApi.getLibraryState(track.uri).setResultCallback { ls ->
                    if (lastTrackUri == track.uri) _state.update { it.copy(saved = ls.isAdded) }
                }
            }
        }
        subscription.setErrorCallback {
            _state.update { it.copy(error = "Связь со Spotify потеряна") }
        }
        sub = subscription
    }

    override fun disconnect() {
        prefs.edit().putBoolean("auto", false).apply()
        teardown()
    }

    private fun teardown() {
        coverRequested.clear()
        shelvesLoading = false
        if (current === this) current = null
        sub?.cancel(); sub = null
        remote?.let { SpotifyAppRemote.disconnect(it) }
        remote = null
        lastTrackUri = null
        handler.removeCallbacks(ticker)
        // Всё, что пришло из Spotify, стираем из памяти
        _state.value = SpotifyState(appInstalled = SpotifyAppRemote.isSpotifyInstalled(activity))
    }

    /** Вызывается при закрытии активности. Если приложение закрыли совсем (не поворот экрана), ставим Spotify на паузу. */
    fun release() {
        val r = remote
        android.util.Log.d("SpotifyRelease", "release finishing=${activity.isFinishing} config=${activity.isChangingConfigurations} connected=${r?.isConnected}")
        if (!activity.isChangingConfigurations && r != null && r.isConnected) {
            r.playerApi.pause().setResultCallback { teardown() }.setErrorCallback { teardown() }
            handler.postDelayed({ teardown() }, 800)
        } else {
            teardown()
        }
    }

    override fun playUri(uri: String) {
        val r = remote ?: return
        ignoreIdle = false
        // Сразу показываем выбранное в мини-плеере: пока Spotify отвечает, плеер не пропадает
        val known = listItems[uri]
        val hist = _state.value.history.firstOrNull { it.uri == uri }
        val name = known?.title ?: hist?.title
        if (name != null) {
            val key = known?.imageUri?.raw ?: hist?.imageKey
            optimisticKey = key
            _state.update { st ->
                st.copy(
                    uri = uri,
                    title = name,
                    artist = known?.subtitle ?: hist?.subtitle,
                    isPaused = false,
                    positionMs = 0L,
                    durationMs = 0L,
                    cover = key?.let { st.covers[it] },
                    // Трек сразу встаёт первым в «Недавно играло», чтобы список не менялся второй раз при ответе Spotify
                    history = if (uri.startsWith("spotify:track:")) {
                        (listOf(SpotifyItem(uri, name, known?.subtitle ?: hist?.subtitle, imageKey = key)) +
                            st.history.filterNot { h -> h.uri == uri }).take(30)
                    } else st.history,
                    error = null,
                )
            }
        }
        r.playerApi.play(uri).setErrorCallback {
            _state.update { it.copy(isPaused = true, error = "Не удалось запустить. Нужен Premium и открытый вход в Spotify.") }
        }
    }

    override fun pause() { remote?.playerApi?.pause() }
    override fun resume() { remote?.playerApi?.resume() }
    override fun next() { remote?.playerApi?.skipNext() }
    override fun previous() { remote?.playerApi?.skipPrevious() }
    override fun seekTo(ms: Long) { remote?.playerApi?.seekTo(ms) }

    override fun toggleSaved() {
        val r = remote ?: return
        val uri = _state.value.uri ?: return
        val add = !_state.value.saved
        _state.update { it.copy(saved = add) }
        // Лайк уходит в Spotify и у нас нигде не хранится
        if (add) r.userApi.addToLibrary(uri) else r.userApi.removeFromLibrary(uri)
    }

    private val coverRequested = HashSet<String>()

    /** Обложка элемента: берём у самого Spotify без изменений, держим только в памяти. */
    private fun requestCover(uri: com.spotify.protocol.types.ImageUri?) {
        val r = remote ?: return
        val u = uri ?: run { android.util.Log.d("SpotifyCover", "нет imageUri"); return }
        val key = u.raw
        if (key.isNullOrBlank()) {
            android.util.Log.d("SpotifyCover", "пустой imageUri")
            return
        }
        if (!coverRequested.add(key)) return
        r.imagesApi.getImage(u, Image.Dimension.MEDIUM)
            .setResultCallback { bmp ->
                _state.update { st -> st.copy(covers = st.covers + (key to bmp.asImageBitmap())) }
            }
            .setErrorCallback { e ->
                coverRequested.remove(key)
                android.util.Log.d("SpotifyCover", "ошибка $key: ${e.message}")
            }
    }

    private val listItems = HashMap<String, com.spotify.protocol.types.ListItem>()

    private fun showItems(title: String, items: Array<com.spotify.protocol.types.ListItem>?) {
        val list = (items ?: emptyArray()).filter { it.uri != null && it.title != null }
        list.forEach { listItems[it.uri] = it; requestCover(it.imageUri) }
        _state.update { st ->
            st.copy(browseTitle = title, browse = list.map {
                SpotifyItem(it.uri, it.title, it.subtitle, it.playable, it.hasChildren, it.imageUri?.raw)
            })
        }
    }

    private var shelvesLoading = false

    private fun toSpotifyItem(li: com.spotify.protocol.types.ListItem) =
        SpotifyItem(li.uri, li.title, li.subtitle, li.playable, li.hasChildren, li.imageUri?.raw)

    /** Каждая верхняя полка Spotify становится отдельным рядом (не больше 12 элементов, лимит Spotify 20). */
    override fun loadRecommended() {
        val r = remote ?: return
        if (shelvesLoading) return
        shelvesLoading = true
        r.contentApi.getRecommendedContentItems(com.spotify.android.appremote.api.ContentApi.ContentType.DEFAULT)
            .setResultCallback { top ->
                val tops = (top.items ?: emptyArray()).filter { it.uri != null && it.title != null }
                tops.forEach { listItems[it.uri] = it }
                val sources = tops.filter { it.hasChildren && !it.playable }.take(8)
                val results = arrayOfNulls<SpotifyShelf>(sources.size)
                var pending = sources.size
                fun publish() {
                    val ordered = results.filterNotNull()
                    if (ordered.isNotEmpty() || pending == 0) {
                        _state.update { st -> st.copy(shelves = ordered) }
                    }
                    if (pending == 0) shelvesLoading = false
                }
                if (sources.isEmpty()) publish()
                sources.forEachIndexed { index, shelf ->
                    r.contentApi.getChildrenOfItem(shelf, 12, 0)
                        .setResultCallback { kids ->
                            val list = (kids.items ?: emptyArray()).filter { it.uri != null && it.title != null }
                            list.forEach { listItems[it.uri] = it; requestCover(it.imageUri) }
                            if (list.isNotEmpty()) results[index] = SpotifyShelf(shelf.title, list.map { toSpotifyItem(it) })
                            pending--
                            publish()
                        }
                        .setErrorCallback {
                            pending--
                            publish()
                        }
                }
            }
            .setErrorCallback {
                shelvesLoading = false
                _state.update { st -> st.copy(shelves = emptyList()) }
            }
    }

    override fun openItem(item: SpotifyItem) {
        val r = remote ?: return
        val li = listItems[item.uri] ?: return
        r.contentApi.getChildrenOfItem(li, 20, 0)
            .setResultCallback { showItems(item.title, it.items) }
            .setErrorCallback { _state.update { st -> st.copy(error = "Не удалось открыть подборку") } }
    }

    override fun closeBrowse() {
        _state.update { it.copy(browse = null, browseTitle = null) }
    }

    override fun openSpotifyApp() {
        activity.packageManager.getLaunchIntentForPackage("com.spotify.music")?.let { activity.startActivity(it) }
    }

    override fun getSpotifyApp() {
        val uri = Uri.parse("market://details?id=com.spotify.music")
        runCatching { activity.startActivity(Intent(Intent.ACTION_VIEW, uri)) }.onFailure {
            activity.startActivity(
                Intent(Intent.ACTION_VIEW, Uri.parse("https://play.google.com/store/apps/details?id=com.spotify.music"))
            )
        }
    }
}

package com.unknowstorm.musicapp.spotify

import android.app.Service
import android.content.Intent
import android.os.Handler
import android.os.IBinder

import android.util.Log

/** Срабатывает, когда Solara смахнули из недавних: ставит Spotify на паузу, даже если у активности не было onDestroy. */
class SpotifyTaskGuard : Service() {
    override fun onBind(intent: Intent?): IBinder? = null

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int = START_NOT_STICKY

    override fun onTaskRemoved(rootIntent: Intent?) {
        Log.d("SpotifyGuard", "task removed, current=${SpotifyRemoteImpl.current != null}")
        SpotifyRemoteImpl.current?.pauseNow()
        Handler(Looper.getMainLooper()).postDelayed({ stopSelf() }, 700)
        super.onTaskRemoved(rootIntent)
    }
}

// Сколько показываем вращение заставки, когда приложение готово: минимум и потолок (например, нет сети).
// Время считается уже после входа логотипа и первой сборки интерфейса.
private const val SPLASH_MIN_MS = 450L
private const val SPLASH_MAX_MS = 1000L

/** Всё тяжёлое, что создаётся в фоне, пока главный поток рисует заставку. */
private class AppBoot(
    val providers: List<MusicProvider>,
    val authRepo: AuthRepository,
)

@Composable
fun App() {
    SolaraTheme {
        Surface(Modifier.fillMaxSize()) {
            // HTTP-клиенты, Supabase и провайдеры создаём в фоновом потоке: главный в это время свободен для анимации
            val boot by produceState<AppBoot?>(initialValue = null) {
                value = withContext(Dispatchers.Default) {
                    AppBoot(
                        providers = listOf<MusicProvider>(
                            JamendoProvider(Secrets.JAMENDO_CLIENT_ID),
                            AudiusProvider(Secrets.AUDIUS_API_KEY),
                            DeezerProvider(),
                        ),
                        authRepo = AuthRepository(),
                    )
                }
            }
            var entered by remember { mutableStateOf(false) }
            var spinning by remember { mutableStateOf(false) }
            var leaving by remember { mutableStateOf(false) }
            var splashDone by remember { mutableStateOf(false) }

            // Тяжёлую первую сборку интерфейса запускаем, когда готов фон и логотип уже вошёл:
            // подвисание (если оно будет) приходится на неподвижную заставку, а не на вращение.
            val ready = if (entered) boot else null

            Box(Modifier.fillMaxSize()) {
                if (ready != null) {
                    AppContent(ready)

                    val authState by ready.authRepo.state.collectAsState()
                    // Вход «готов», когда понятно, гость это или вошедший пользователь с загруженным профилем
                    val authSettled = when (val s = authState) {
                        AuthState.Loading -> false
                        is AuthState.SignedIn -> s.profile != null
                        else -> true
                    }
                    val settled by rememberUpdatedState(authSettled)
                    LaunchedEffect(Unit) {
                        // два кадра: тяжёлый первый кадр интерфейса уже нарисован, вращение стартует с чистого листа
                        withFrameNanos { }
                        withFrameNanos { }
                        spinning = true
                        delay(SPLASH_MIN_MS)
                        withTimeoutOrNull(SPLASH_MAX_MS - SPLASH_MIN_MS) {
                            snapshotFlow { settled }.first { it }
                        }
                        leaving = true
                    }
                }
                if (!splashDone) {
                    SplashScreen(
                        spinning = spinning,
                        leaving = leaving,
                        onEntered = { entered = true },
                        onFinished = { splashDone = true },
                    )
                }
            }
        }
    }
}

@Composable
private fun AppContent(boot: AppBoot) {
    val providers = boot.providers
    val authRepo = boot.authRepo
    val audio = rememberAudioBackend()
    val likes = remember { LikesRepository(authRepo) }
    val authState by authRepo.state.collectAsState()
    val pending by authRepo.pending.collectAsState()
    var showPlayer by remember { mutableStateOf(false) }
    var showAuth by remember { mutableStateOf(false) }
    val miniAnchor = remember { MiniPlayerAnchor() }
    val spotify = LocalSpotifyRemote.current
    val spotifyMini = remember { SpotifyMiniState() }
    val focusManager = LocalFocusManager.current
    // Снимаем фокус с поля поиска, иначе его «капля» остаётся поверх плеера
    LaunchedEffect(showPlayer) { if (showPlayer) focusManager.clearFocus() }
    // Вошли (в том числе через Google после возврата из браузера): закрываем экран входа
    LaunchedEffect(authState) {
        if (authState is AuthState.SignedIn) showAuth = false
    }
    CompositionLocalProvider(
        LocalMiniPlayerAnchor provides miniAnchor,
        LocalSpotifyMini provides spotifyMini,
    ) {
        Box(Modifier.fillMaxSize()) {
            MainScreen(
                providers = providers,
                audio = audio,
                authRepo = authRepo,
                likes = likes,
                onOpenPlayer = { showPlayer = true },
                onOpenAuth = { showAuth = true },
            )
            // Мини-плеер и полный плеер — один компонент: плавно перетекает из одного в другой
            ExpandablePlayer(
                audio = audio,
                likes = likes,
                expanded = showPlayer,
                onExpandedChange = { showPlayer = it },
                // Гость нажал ♡: сворачиваем плеер и открываем вход
                onNeedAuth = {
                    showPlayer = false
                    showAuth = true
                },
            )
            // Мини-плеер Spotify: отдельный компонент, не связан с нашим AudioBackend-плеером
            if (spotify != null) {
                SpotifyExpandablePlayer(remote = spotify, audio = audio)
            }
            // Экран входа и регистрации поверх всего
            if (showAuth) {
                PlatformBackHandler(enabled = true, onBack = { showAuth = false })
                AuthScreen(repo = authRepo, onClose = { showAuth = false })
            }
            // После регистрации: «Ожидаем подтверждение письма».
            // Закрывается сам, когда письмо подтверждено и пользователь вошёл.
            pending?.let { p ->
                AwaitEmailScreen(
                    repo = authRepo,
                    pending = PendingSignUp(p.email, p.password),
                    onBack = { authRepo.cancelPending() },
                )
            }
            // Первый вход: ник, аватарка, сервисы, синхронизация палитры.
            // Закрывается сам, когда профиль сохранён (onboarded = true).
            val signedIn = authState as? AuthState.SignedIn
            val profile = signedIn?.profile
            if (profile != null && !profile.onboarded) {
                OnboardingScreen(repo = authRepo, profile = profile)
            }
        }
    }
}

/** Запрос подборки с повторами: на холодном старте сеть часто ещё не готова (DNS, таймаут). */
private suspend fun searchWithRetry(provider: MusicProvider, query: String): List<Track> {
    var last: Exception? = null
    repeat(3) { attempt ->
        try {
            return provider.search(query)
        } catch (e: CancellationException) {
            throw e
        } catch (e: Exception) {
            last = e
            delay(800L * (attempt + 1))
        }
    }
    throw last ?: IllegalStateException("search failed")
}

/** Лента одного источника. Живёт на уровне MainScreen, чтобы не грузиться заново при смене вкладок. */
class HomeFeed(
    val service: Service,
    private val provider: MusicProvider,
    private val mixes: List<Pair<String, String>>,
) {
    var sections by mutableStateOf<List<FeedSection>>(emptyList())
        private set
    var loading by mutableStateOf(false)
        private set
    var error by mutableStateOf<String?>(null)
        private set

    suspend fun loadIfNeeded() {
        if (sections.isEmpty() && !loading) {
            load()
            // Если сеть так и не ответила, пробуем ещё пару раз сами, без нажатия «Повторить»
            var tries = 0
            while (sections.isEmpty() && error != null && tries < 2) {
                tries++
                delay(4000)
                load()
            }
        }
    }

    suspend fun load() {
        loading = true
        error = null
        try {
            var lastError: String? = null
            val loaded = coroutineScope {
                mixes.map { (title, query) ->
                    async {
                        try {
                            FeedSection(title, searchWithRetry(provider, query).take(12))
                        } catch (e: CancellationException) {
                            throw e
                        } catch (e: Exception) {
                            println("Solara: feed ${service.title}/$title failed: ${e.stackTraceToString()}")
                            lastError = e.message
                            null
                        }
                    }
                }.awaitAll()
            }.filterNotNull().filter { it.tracks.isNotEmpty() }
            sections = loaded
            if (loaded.isEmpty() && lastError != null) error = "Ошибка загрузки: $lastError"
        } finally {
            loading = false
        }
    }
}

private data class FeedRow(val key: String, val title: String, val service: Service?, val tracks: List<Track>)

private fun <T> roundRobin(lists: List<List<T>>): List<T> {
    val result = mutableListOf<T>()
    val longest = lists.maxOfOrNull { it.size } ?: 0
    for (i in 0 until longest) {
        for (list in lists) list.getOrNull(i)?.let(result::add)
    }
    return result
}

/** Собирает ряды главной. На вкладке «Все» источники чередуются, а сверху идёт общий микс. */
private fun buildRows(feeds: List<HomeFeed>, mixed: Boolean): List<FeedRow> {
    val perFeed = feeds.map { feed ->
        feed.sections.map { FeedRow("${feed.service.name}:${it.title}", it.title, feed.service, it.tracks.filterNot { t -> DeadTracks.isDead(t) }) }
    }
    val rows = roundRobin(perFeed)
    if (!mixed) return rows
    val firstRows = perFeed.mapNotNull { it.firstOrNull() }
    if (firstRows.size < 2) return rows
    val mix = roundRobin(firstRows.map { it.tracks }).take(16)
    return listOf(FeedRow("mix", "Микс из всех источников", Service.ALL, mix)) + rows
}

/**
 * Лайки хранят только метаданные, поэтому ссылку на звук берём заново у источника:
 * у Deezer она протухает. Что не удалось получить, пропускаем.
 */
internal suspend fun playLiked(
    tracks: List<Track>,
    start: Track,
    providers: List<MusicProvider>,
    audio: AudioBackend,
) {
    // Окно вокруг выбранного трека: так работает и выбор из хвоста длинного списка
    val from = tracks.indexOfFirst { it.provider == start.provider && it.id == start.id }.coerceAtLeast(0)
    val window = tracks.drop((from - 5).coerceAtLeast(0)).take(30)
    val resolved = coroutineScope {
        window.map { track ->
            async {
                val provider = providers.firstOrNull { it.id == track.provider } ?: return@async null
                val url = try {
                    withTimeoutOrNull(6_000) { provider.resolveStream(track) }
                } catch (e: CancellationException) {
                    throw e
                } catch (e: Exception) {
                    null
                }
                url?.takeIf { it.isNotBlank() }?.let { track.copy(streamUrl = it) }
            }
        }.awaitAll()
    }.filterNotNull()
    val index = resolved.indexOfFirst { it.provider == start.provider && it.id == start.id }
    if (index >= 0) audio.playQueue(resolved, index)
}

@Composable
fun MainScreen(
    providers: List<MusicProvider>,
    audio: AudioBackend,
    authRepo: AuthRepository,
    likes: LikesRepository,
    onOpenPlayer: () -> Unit,
    onOpenAuth: () -> Unit,
) {
    var tab by remember { mutableStateOf(Tab.HOME) }
    var service by remember { mutableStateOf(Service.ALL) }
    var showLikes by remember { mutableStateOf(false) }

    val feeds = remember(providers) {
        providers.mapNotNull { provider ->
            val s = Service.entries.firstOrNull { it.name == provider.id.toString() }
            s?.let { HomeFeed(it, provider, defaultMixes(it)) }
        }
    }
    val shownFeeds = remember(feeds, service) {
        if (service == Service.ALL) feeds.filter { it.service.kind == com.unknowstorm.musicapp.model.PlaybackKind.IN_APP } else feeds.filter { it.service == service }
    }

    val ytScope = rememberCoroutineScope()
    val session = remember { YouTubeSession(ytScope) }
    var rootOrigin by remember { mutableStateOf(Offset.Zero) }
    var rootSize by remember { mutableStateOf(IntSize.Zero) }
    // Большой плеер YouTube развёрнут полностью: нижняя панель навигации прячется (портрет и альбом).
    val hideNavBar by remember {
        derivedStateOf {
            session.video != null && session.progress >= 0.999f &&
                tab == Tab.HOME && service == Service.YOUTUBE
        }
    }
    val playback by audio.state.collectAsState()
    val spotify = LocalSpotifyRemote.current
    // Наша музыка или YouTube заиграли: Spotify на паузу
    LaunchedEffect(playback.isPlaying) { if (playback.isPlaying) spotify?.pause() }
    LaunchedEffect(session.video) { if (session.video != null) spotify?.pause() }
    // Включили нашу музыку: видео YouTube закрывается, два звука одновременно не играют
    LaunchedEffect(playback.mediaId) {
        if (playback.mediaId != null && session.video != null) session.close()
    }

    Box(
        Modifier
            .fillMaxSize()
            .onGloballyPositioned { rootOrigin = it.positionInWindow() }
            .onSizeChanged { rootSize = it }
    ) {
    // Делаем фон колонки прозрачным (чтобы не было черных полос)
    Column(Modifier.fillMaxSize().background(Color.Transparent)) {
        Box(Modifier.weight(1f).fillMaxWidth()) {
            when (tab) {
                Tab.HOME -> if (service == Service.SPOTIFY) {
                    SpotifyScreen(
                        remote = spotify,
                        onServiceChange = { service = it; showLikes = false },
                        onOpenLikes = { service = Service.ALL; showLikes = true },
                        onBack = { service = Service.ALL },
                        onBeforePlay = { audio.stop(); session.close() },
                    )
                } else if (service == Service.YOUTUBE) {
                    YouTubeScreen(
                        audio = audio,
                        session = session,
                        onBack = { service = Service.ALL },
                        onServiceChange = { service = it },
                    )
                } else {
                    HomeScreen(shownFeeds, service, { service = it; showLikes = false }, audio, likes, providers, showLikes, { showLikes = true }, onOpenAuth)
                }
                Tab.SEARCH -> TrackListScreen(providers, audio, session, onOpenSpotify = { tab = Tab.HOME; service = Service.SPOTIFY })
                Tab.QUEUE -> InfoScreen(
                    "Очередь",
                    "Здесь будет текущая очередь: удаление треков, шафл и повтор списка. Следующий этап.",
                )
                Tab.SETTINGS -> SettingsScreen(authRepo, onOpenAuth)
            }
        }

        // Место под мини-плеер Spotify (рисует его SpotifyExpandablePlayer)
        SpotifyMiniSlot(hidden = playback.title != null)

        // Мини-плеер лежит над нижней панелью
        NowPlayingBar(
            audio = audio,
            onOpen = onOpenPlayer,
            // Убираем лишние отступы, чтобы плеер прилегал вплотную
            modifier = Modifier.padding(horizontal = 8.dp, vertical = 8.dp)
        )

        // ИСПРАВЛЕНИЕ: Убираем тень у NavigationBar и делаем его прозрачным
        if (!hideNavBar) NavigationBar(
            containerColor = MaterialTheme.colorScheme.background,
            tonalElevation = 0.dp, // Убирает черную тень!
            modifier = Modifier.background(Color.Transparent)
        ) {
            Tab.entries.forEach { item ->
                NavigationBarItem(
                    selected = tab == item,
                    onClick = { tab = item },
                    icon = {
                        when (item) {
                            Tab.HOME -> HomeIcon()
                            Tab.SEARCH -> SearchIcon()
                            Tab.QUEUE -> QueueIcon()
                            Tab.SETTINGS -> SettingsIcon()
                        }
                    },
                    label = { Text(item.label) },
                )
            }
        }
    }

        // Единственный плеер YouTube: на экране YouTube большой, в остальных местах плавающее окно
        YouTubeOverlay(
            session = session,
            containerOrigin = rootOrigin,
            containerSize = rootSize,
            onExpand = {
                tab = Tab.HOME
                service = Service.YOUTUBE
                session.docked = true
            },
        )
    }

@Composable
fun OnboardingScreen(
    repo: AuthRepository,
    profile: Profile,
    modifier: Modifier = Modifier,
    onDone: (() -> Unit)? = null,
) {
    val scope = rememberCoroutineScope()
    var nick by remember { mutableStateOf(profile.username.orEmpty()) }
    var syncPalette by remember { mutableStateOf(profile.syncPalette) }
    var services by remember { mutableStateOf(profile.services.toSet()) }
    var busy by remember { mutableStateOf(false) }
    var error by remember { mutableStateOf<String?>(null) }

    // Аватарка: уже сохранённое фото, только что выбранное или «убрать фото»
    val savedImage = rememberAvatarImage(repo, profile.avatar)
    var pickedBytes by remember { mutableStateOf<ByteArray?>(null) }
    var removePhoto by remember { mutableStateOf(false) }
    val pickedImage = remember(pickedBytes) { pickedBytes?.let { decodeImageBitmap(it) } }
    val shownImage = if (removePhoto) null else pickedImage ?: savedImage
    val pickPhoto = rememberAvatarPicker { bytes ->
        pickedBytes = bytes
        removePhoto = false
        error = null
    }

    val nickValid = NickRegex.matches(nick.trim())
    val cardColors = CardDefaults.cardColors(containerColor = MaterialTheme.colorScheme.surfaceContainer)
    val transparentItem = ListItemDefaults.colors(containerColor = Color.Transparent)

    Surface(modifier.fillMaxSize(), color = MaterialTheme.colorScheme.surface) {
        Box(
            Modifier.fillMaxSize().windowInsetsPadding(WindowInsets.systemBars).imePadding(),
            contentAlignment = Alignment.TopCenter,
        ) {
            Column(
                Modifier
                    .widthIn(max = 480.dp)
                    .fillMaxWidth()
                    .verticalScroll(rememberScrollState())
                    .padding(horizontal = 24.dp, vertical = 16.dp),
                verticalArrangement = Arrangement.spacedBy(16.dp),
            ) {
                if (onDone != null) {
                    TextButton(onClick = onDone, enabled = !busy) { Text("← Назад") }
                    Text("Профиль", style = MaterialTheme.typography.headlineLarge)
                } else {
                    Text("Почта подтверждена", style = MaterialTheme.typography.labelLarge, color = MaterialTheme.colorScheme.primary)
                    Text("Расскажи о себе", style = MaterialTheme.typography.headlineLarge)
                }

                // Аватарка: фото или первая буква ника на цвете палитры
                Column(
                    Modifier.fillMaxWidth(),
                    horizontalAlignment = Alignment.CenterHorizontally,
                    verticalArrangement = Arrangement.spacedBy(12.dp),
                ) {
                    UserAvatar(
                        name = nick,
                        size = 96.dp,
                        image = shownImage,
                        modifier = Modifier.clip(CircleShape).clickable(enabled = !busy) { pickPhoto() },
                    )
                    Row(horizontalArrangement = Arrangement.spacedBy(8.dp), verticalAlignment = Alignment.CenterVertically) {
                        FilledTonalButton(onClick = pickPhoto, enabled = !busy) {
                            Text(if (shownImage == null) "Загрузить фото" else "Другое фото")
                        }
                        if (shownImage != null) {
                            TextButton(
                                onClick = { pickedBytes = null; removePhoto = true },
                                enabled = !busy,
                            ) { Text("Убрать") }
                        }
                    }
                }

                // Ник
                OutlinedTextField(
                    value = nick,
                    onValueChange = { nick = it.take(20); error = null },
                    label = { Text("Ник") },
                    singleLine = true,
                    isError = nick.isNotEmpty() && !nickValid,
                    supportingText = { Text("3–20 символов: латинские буквы, цифры и _") },
                    shape = RoundedCornerShape(16.dp),
                    modifier = Modifier.fillMaxWidth(),
                )

                // Синхронизация палитры
                Card(Modifier.fillMaxWidth(), shape = RoundedCornerShape(24.dp), colors = cardColors) {
                    ListItem(
                        headlineContent = { Text("Синхронизировать палитру цветов") },
                        supportingContent = { Text("Со всеми твоими устройствами через облако") },
                        trailingContent = { Switch(checked = syncPalette, onCheckedChange = { syncPalette = it }) },
                        colors = transparentItem,
                    )
                }

                // Сервисы: квадратные кнопки со скруглёнными углами
                Text("Сервисы", style = MaterialTheme.typography.titleMedium)
                Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(12.dp)) {
                    ServiceChoices.forEach { (key, title) ->
                        ServiceTile(
                            title = title,
                            selected = key in services,
                            onClick = { services = if (key in services) services - key else services + key },
                            modifier = Modifier.weight(1f),
                        )
                    }
                }
                Text(
                    "Вход в аккаунт сервиса появится в обновлении",
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                )

                error?.let {
                    Text(it, color = MaterialTheme.colorScheme.error, style = MaterialTheme.typography.bodyMedium)
                }

                Button(
                    onClick = {
                        scope.launch {
                            busy = true; error = null
                            val picked = pickedBytes
                            val avatarUpdate = when {
                                picked != null -> AvatarUpdate.Upload(picked)
                                removePhoto -> AvatarUpdate.Remove
                                else -> AvatarUpdate.Keep
                            }
                            // При успехе профиль помечается заполненным, и App сам закроет этот экран
                            repo.saveOnboarding(nick.trim(), avatarUpdate, syncPalette, services)
                                .onSuccess { onDone?.invoke() }
                                .onFailure { error = it.message }
                            busy = false
                        }
                    },
                    enabled = nickValid && !busy,
                    modifier = Modifier.fillMaxWidth().height(56.dp),
                    shape = RoundedCornerShape(28.dp),

                    if (busy) {
                        CircularProgressIndicator(

                            color = MaterialTheme.colorScheme.onPrimary,
                        )
                    } else {
                        Text("Готово")
                    }
                }

                if (onDone == null) {
                    TextButton(
                        onClick = { scope.launch { repo.signOut() } },
                        enabled = !busy,
                        modifier = Modifier.align(Alignment.CenterHorizontally),
                    ) { Text("Выйти из аккаунта") }
                }
            }
        }
    }

/**
 * Мини-плеер и полный плеер Spotify: тот же морфинг, что у ExpandablePlayer,
 * но отдельный компонент: ничего общего с AudioBackend и нашими источниками.
 * Ставить в корневой Box поверх основного контента.
 */
@Composable
fun SpotifyExpandablePlayer(remote: SpotifyRemote, audio: AudioBackend, modifier: Modifier = Modifier) {
    val ui = LocalSpotifyMini.current
    val anchor = ui.anchor
    val ownAnchor = LocalMiniPlayerAnchor.current
    val stateHolder = remote.state.collectAsState()
    val audioHolder = audio.state.collectAsState()
    // Сырое условие. Spotify при смене трека может на секунду прислать «пустое» состояние
    val rawHasTrack by remember {
        derivedStateOf {
            val st = stateHolder.value
            st.connected && st.premium != false && st.title != null &&
                audioHolder.value.title == null &&
                !(ui.dismissedUri != null && ui.dismissedUri == st.uri && st.isPaused)
        }
    }
    val audioBusy by remember { derivedStateOf { audioHolder.value.title != null } }
    // Последнее состояние с названием трека: его показываем, пока длится короткий разрыв
    val held = remember { mutableStateOf(stateHolder.value) }
    LaunchedEffect(Unit) {
        snapshotFlow { stateHolder.value }.collect { if (it.title != null) held.value = it }
    }
    // Spotify на миг присылает пустое состояние при смене трека: держим плеер до 1,2 с, но только в неявных случаях
    var everShown by remember { mutableStateOf(false) }
    var graceExpired by remember { mutableStateOf(false) }
    val explicitHide by remember {
        derivedStateOf {
            val st = stateHolder.value
            // Наш плеер должен сначала занять своё место, и только потом Spotify скрывается: без разрыва между ними
            (audioBusy && ownAnchor.bounds != null) || !st.connected || st.premium == false ||
                (ui.dismissedUri != null && ui.dismissedUri == st.uri && st.isPaused)
        }
    }
    LaunchedEffect(rawHasTrack) {
        if (rawHasTrack) {
            everShown = true
            graceExpired = false
        } else {
            delay(1200)
            graceExpired = true
        }
    }
    // Считаем прямо в композиции: плеер скрывается и появляется в том же кадре, что и смена источника
    val hasTrack = rawHasTrack || (everShown && !graceExpired && !explicitHide)
    val expanded = ui.expanded
    val scope = rememberCoroutineScope()

    SideEffect { ui.shown = hasTrack }
    LaunchedEffect(Unit) {
        snapshotFlow { Triple(rawHasTrack, audioHolder.value.title != null, stateHolder.value.title) }.collect {
            println("Solara: timeline spotifyMini=${it.first} audioTitle=${it.second} spTitle=${it.third} spPaused=${stateHolder.value.isPaused} audioPlaying=${audioHolder.value.isPlaying}")
        }
    }
    DisposableEffect(Unit) { onDispose { ui.shown = false } }
    // Снова нажали play: закрытый ранее трек опять показываем
    LaunchedEffect(Unit) {
        snapshotFlow { stateHolder.value.isPaused }.collect { paused -> if (!paused) ui.dismissedUri = null }
    }

    val progress = remember { mutableFloatStateOf(if (expanded) 1f else 0f) }
    var animJob by remember { mutableStateOf<Job?>(null) }
    val settling = remember { floatArrayOf(-1f) }

    fun settle(target: Float, velocity: Float = 0f) {
        settling[0] = target
        animJob?.cancel()
        animJob = scope.launch {
            animate(
                initialValue = progress.floatValue,
                targetValue = target,
                initialVelocity = velocity,
                animationSpec = spring(
                    dampingRatio = Spring.DampingRatioNoBouncy,
                    stiffness = Spring.StiffnessMediumLow,
                ),
            ) { v, _ -> progress.floatValue = v.coerceIn(0f, 1f) }
            settling[0] = -1f
        }
    }

    LaunchedEffect(expanded) {
        val target = if (expanded) 1f else 0f
        if (settling[0] != target && progress.floatValue != target) settle(target)
    }

    LaunchedEffect(hasTrack) {
        if (!hasTrack) {
            animJob?.cancel()
            settling[0] = -1f
            progress.floatValue = 0f
            ui.expanded = false
        }
    }
    if (!hasTrack) return

    PlatformBackHandler(enabled = expanded, onBack = { ui.expanded = false })

    var origin by remember { mutableStateOf(Offset.Zero) }
    var miniOffsetX by remember { mutableFloatStateOf(0f) }
    var dismissJob by remember { mutableStateOf<Job?>(null) }

    fun animateMiniX(to: Float, after: () -> Unit = {}) {
        dismissJob?.cancel()
        dismissJob = scope.launch {
            animate(initialValue = miniOffsetX, targetValue = to) { v, _ -> miniOffsetX = v }
            after()
        }
    }

    fun dismissMini() {
        ui.dismissedUri = stateHolder.value.uri
        remote.pause()
    }

    BoxWithConstraints(
        modifier
            .fillMaxSize()
            .onGloballyPositioned { origin = it.positionInRoot() }
    ) {
        val density = LocalDensity.current
        val fullW = constraints.maxWidth.toFloat()
        val fullH = constraints.maxHeight.toFloat()
        val fullWidthDp = maxWidth
        val fullHeightDp = maxHeight

        val topInset = WindowInsets.statusBars.asPaddingValues().calculateTopPadding()
        val bottomInset = WindowInsets.navigationBars.asPaddingValues().calculateBottomPadding()

        val fbMargin = with(density) { 8.dp.toPx() }
        val fbBottom = with(density) { (bottomInset + 8.dp).toPx() }
        val miniHpx = with(density) { MiniHeight.toPx() }
        fun miniRect(): Rect {
            val b = anchor.bounds
            return if (b != null) {
                Rect(b.left - origin.x, b.top - origin.y, b.right - origin.x, b.bottom - origin.y)
            } else {
                Rect(fbMargin, fullH - fbBottom - miniHpx, fullW - fbMargin, fullH - fbBottom)
            }
        }
        val miniNow = miniRect()
        val miniWidthDp = with(density) { miniNow.width.toDp() }

        val layoutDirection = LocalLayoutDirection.current
        val sideInsets = WindowInsets.systemBars.union(WindowInsets.displayCutout)
        val leftInset = with(density) { sideInsets.getLeft(density, layoutDirection).toDp() }
        val rightInset = with(density) { sideInsets.getRight(density, layoutDirection).toDp() }
        val landscape = fullW > fullH

        val coverTop: Dp
        val coverSize: Dp
        val coverLeft: Dp
        if (landscape) {
            val availH = maxHeight - topInset - bottomInset
            coverSize = maxOf(120.dp, minOf(availH - 24.dp, maxWidth * 0.42f))
            coverTop = topInset + (availH - coverSize) / 2f
            coverLeft = leftInset + 32.dp
        } else {
            coverTop = topInset + 32.dp + SpCoverDownShift
            coverSize = maxOf(160.dp, minOf(maxWidth - SpScreenSideMargin * 2f, maxHeight - coverTop - bottomInset - SpBelowCoverBlock))
            coverLeft = (maxWidth - coverSize) / 2f
        }
        val coverTopPx = with(density) { coverTop.toPx() }
        val bigX = with(density) { coverLeft.toPx() }
        val belowCoverStart = coverTop + coverSize + 20.dp
        val panelStart = coverLeft + coverSize + 32.dp

        val miniCornerPx = with(density) { SpMiniCorner.toPx() }
        val shadowPx = with(density) { 3.dp.toPx() }
        val miniColor = MaterialTheme.colorScheme.secondaryContainer
        val fullColor = MaterialTheme.colorScheme.surface

        // Запоминаем в момент расстановки, стоит ли плеер на измеренном месте: до этого он невидим
        val placedOnAnchor = remember { mutableStateOf(false) }
        val miniVisible by remember { derivedStateOf { progress.floatValue < 0.999f } }
        val fullVisible by remember { derivedStateOf { progress.floatValue > 0.001f } }

        val dragState = rememberDraggableState { delta ->
            val travel = miniRect().top.coerceAtLeast(1f)
            progress.floatValue = (progress.floatValue - delta / travel).coerceIn(0f, 1f)
        }

        Box(
            modifier = Modifier
                .offset {
                    val m = miniRect()
                    val p = progress.floatValue
                    IntOffset(lerp(m.left, 0f, p).roundToInt(), lerp(m.top, 0f, p).roundToInt())
                }
                .layout { measurable, _ ->
                    placedOnAnchor.value = anchor.bounds != null
                    val m = miniRect()
                    val p = progress.floatValue
                    val w = lerp(m.width, fullW, p).roundToInt().coerceAtLeast(0)
                    val h = lerp(m.height, fullH, p).roundToInt().coerceAtLeast(0)
                    val placeable = measurable.measure(Constraints.fixed(w, h))
                    layout(w, h) { placeable.place(0, 0) }
                }
                .graphicsLayer {
                    val p = progress.floatValue
                    translationX = miniOffsetX
                    // Пока место под мини-плеер не измерено, не рисуем его: иначе он мелькает поверх нижней панели
                    alpha = if (!placedOnAnchor.value && p < 0.01f) 0f else 1f
                    shape = RoundedCornerShape(CornerSize(lerp(miniCornerPx, 0f, p)))
                    clip = true
                    shadowElevation = lerp(shadowPx, 0f, p)
                }
                .drawBehind { drawRect(lerpColor(miniColor, fullColor, progress.floatValue)) }
                .draggable(
                    state = dragState,
                    orientation = Orientation.Vertical,
                    onDragStarted = {
                        animJob?.cancel()
                        settling[0] = -1f
                    },
                    onDragStopped = { velocity ->
                        val p = progress.floatValue
                        val travel = miniRect().top.coerceAtLeast(1f)
                        val target = when {
                            velocity > 1500f -> 0f
                            velocity < -1500f -> 1f
                            p < 0.5f -> 0f
                            else -> 1f
                        }
                        settle(target, -velocity / travel)
                        ui.expanded = target == 1f
                    },
                )
                .then(
                    if (expanded) {
                        Modifier
                    } else {
                        Modifier
                            .pointerInput(Unit) {
                                detectHorizontalDragGestures(
                                    onDragStart = { dismissJob?.cancel() },
                                    onDragEnd = {
                                        val m = miniRect()
                                        if (abs(miniOffsetX) > m.width * 0.35f) {
                                            animateMiniX(if (miniOffsetX > 0f) fullW else -fullW) {
                                                dismissMini()
                                                miniOffsetX = 0f
                                            }
                                        } else {
                                            animateMiniX(0f)
                                        }
                                    },
                                    onDragCancel = { animateMiniX(0f) },
                                    onHorizontalDrag = { _, dx -> miniOffsetX += dx },
                                )
                            }
                            .clickable { ui.expanded = true }
                    }
                )
        ) {
            if (miniVisible) {
                SpMiniContent(remote, held, { progress.floatValue }, miniWidthDp) { dismissMini() }
            }
            if (fullVisible) {
                SpFullContent(
                    remote = remote,
                    stateHolder = held,
                    expandProgress = { progress.floatValue },
                    width = fullWidthDp,
                    height = fullHeightDp,
                    topInset = topInset,
                    bottomInset = bottomInset,
                    belowCoverStart = belowCoverStart,
                    landscape = landscape,
                    panelStart = panelStart,
                    rightInset = rightInset,
                )
            }
            SpSharedCover(
                stateHolder = held,
                progress = { progress.floatValue },
                coverSize = coverSize,
                bigX = bigX,
                bigY = coverTopPx,
            )
        }
    }
}

    remote: SpotifyRemote,

    progress: () -> Float,
    width: Dp,
    onClose: () -> Unit,

                    Text(it, style = MaterialTheme.typography.bodyMedium, maxLines = 1, overflow = TextOverflow.Ellipsis)
                }
            }
            Spacer(Modifier.width(6.dp))
            SpotifyIconChip(24.dp)
            Spacer(Modifier.width(8.dp))
            IconButton(
                onClick = { if (s.isPaused) remote.resume() else remote.pause() },

            IconButton(onClick = onClose, modifier = Modifier.size(40.dp)) {
                CloseIcon(Modifier.size(20.dp), MaterialTheme.colorScheme.onSurfaceVariant)
            }
        }
    }
}

@Composable
private fun SpSharedCover(
    stateHolder: State<SpotifyState>,
    progress: () -> Float,
    coverSize: Dp,
    bigX: Float,
    bigY: Float,
) {
    val density = LocalDensity.current
    val cover by remember { derivedStateOf { stateHolder.value.cover } }

    val sizePx = with(density) { coverSize.toPx() }
    val miniSizePx = with(density) { SpMiniCoverSize.toPx() }
    val miniXY = with(density) { SpMiniInset.toPx() }
    val cornerPx = with(density) { SpCoverCorner.toPx() }

    Box(
        Modifier
            .wrapContentSize(Alignment.TopStart, unbounded = true)
            .size(coverSize)
            .graphicsLayer {
                val p = progress()
                val sc = lerp(miniSizePx / sizePx, 1f, p)
                transformOrigin = TransformOrigin(0f, 0f)
                scaleX = sc
                scaleY = sc
                translationX = lerp(miniXY, bigX, p)
                translationY = lerp(miniXY, bigY, p)
            }
    ) {
        Box(
            Modifier
                .fillMaxSize()
                .graphicsLayer {
                    // 4 dp на экране при любом масштабе
                    val sc = lerp(miniSizePx / sizePx, 1f, progress())
                    shape = RoundedCornerShape(CornerSize(cornerPx / sc))
                    clip = true
                }
                .background(MaterialTheme.colorScheme.surfaceVariant)
        ) {
            cover?.let { Image(it, null, Modifier.fillMaxSize(), contentScale = ContentScale.Fit) }
        }
    }
}

@Composable
private fun SpFullContent(
    remote: SpotifyRemote,
    stateHolder: State<SpotifyState>,
    expandProgress: () -> Float,
    width: Dp,
    height: Dp,
    topInset: Dp,
    bottomInset: Dp,
    belowCoverStart: Dp,
    landscape: Boolean,
    panelStart: Dp,
    rightInset: Dp,
) {
    val s by stateHolder
    var dragValue by remember { mutableStateOf<Float?>(null) }
    var showDetails by remember { mutableStateOf(false) }
    // Сообщение при лайке по правилам Spotify; пропадает само
    var likeNote by remember { mutableStateOf<String?>(null) }
    LaunchedEffect(likeNote) { if (likeNote != null) { delay(2000); likeNote = null } }

    val trackProgress = if (s.durationMs > 0) (s.positionMs.toFloat() / s.durationMs).coerceIn(0f, 1f) else 0f
    val shownProgress = dragValue ?: trackProgress
    val shownPositionMs = dragValue?.let { (it * s.durationMs).toLong() } ?: s.positionMs
    val active = MaterialTheme.colorScheme.primary
    val inactive = MaterialTheme.colorScheme.onSurfaceVariant

    CompositionLocalProvider(LocalContentColor provides MaterialTheme.colorScheme.onSurface) {
        Box(
            Modifier
                .wrapContentSize(Alignment.TopStart, unbounded = true)
                .size(width = width, height = height)
                .graphicsLayer { alpha = ((expandProgress() - 0.3f) / 0.5f).coerceIn(0f, 1f) }
        ) {
            Box(
                Modifier
                    .align(Alignment.TopCenter)
                    .padding(top = topInset + 8.dp)
                    .width(40.dp)
                    .height(4.dp)
                    .clip(CircleShape)
                    .background(MaterialTheme.colorScheme.onSurfaceVariant.copy(alpha = 0.3f))
            )
            // Логотип Spotify на экране с его контентом
            SpotifyLogoChip(24.dp, Modifier.align(Alignment.TopCenter).padding(top = topInset + 20.dp))

            Column(
                modifier = if (landscape) {
                    Modifier
                        .fillMaxSize()
                        .padding(start = panelStart, end = rightInset + 32.dp, top = topInset + 16.dp, bottom = bottomInset + 8.dp)
                } else {
                    Modifier.fillMaxSize().padding(horizontal = SpScreenSideMargin).padding(bottom = bottomInset)
                },
                verticalArrangement = if (landscape) Arrangement.Center else Arrangement.Top,
            ) {
                if (!landscape) Spacer(Modifier.height(belowCoverStart))

                Text(
                    s.title ?: "",
                    style = if (landscape) MaterialTheme.typography.titleLarge else MaterialTheme.typography.headlineSmall,
                    maxLines = if (landscape) 1 else 2,
                    overflow = TextOverflow.Ellipsis,
                    modifier = Modifier.fillMaxWidth(),
                )
                Text(
                    s.artist ?: "",
                    style = MaterialTheme.typography.bodyLarge,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    maxLines = 1,
                    overflow = TextOverflow.Ellipsis,
                    modifier = Modifier.fillMaxWidth(),
                )
                s.error?.let {
                    Text(it, color = MaterialTheme.colorScheme.error, style = MaterialTheme.typography.bodySmall)
                }

                Row(
                    Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically,
                ) {
                    IconButton(onClick = { showDetails = true }) { MoreIcon() }
                    SpotifyOpenButton(remote::openSpotifyApp)
                    // «+» кладёт трек в Liked Songs самого Spotify, у нас ничего не хранится
                    IconButton(onClick = {
                        likeNote = if (!s.saved) "Added to Liked Songs" else "Removed from Liked Songs"
                        remote.toggleSaved()
                    }) {
                        Image(
                            painterResource(if (s.saved) Res.drawable.spotify_liked else Res.drawable.spotify_like),
                            contentDescription = if (s.saved) "Liked" else "Like",
                            modifier = Modifier.size(32.dp).clip(CircleShape).background(if (!s.saved && MaterialTheme.colorScheme.surface.luminance() >= 0.179f) Color.Black else Color.Transparent),
                        )
                    }
                }

                likeNote?.let {
                    Text(
                        it,
                        style = MaterialTheme.typography.labelMedium,
                        color = active,
                        textAlign = androidx.compose.ui.text.style.TextAlign.End,
                        modifier = Modifier.fillMaxWidth(),
                    )
                }
                Box(Modifier.fillMaxWidth().height(40.dp), contentAlignment = Alignment.Center) {
                    LinearWavyProgressIndicator(
                        progress = { shownProgress },
                        amplitude = { if (!s.isPaused) 1f else 0f },
                        modifier = Modifier.fillMaxWidth().padding(horizontal = 12.dp),
                    )
                    Slider(
                        value = shownProgress,
                        onValueChange = { dragValue = it },
                        onValueChangeFinished = {
                            dragValue?.let { remote.seekTo((it * s.durationMs).toLong()) }
                            dragValue = null
                        },
                        enabled = s.durationMs > 0,
                        modifier = Modifier.fillMaxWidth().alpha(0f),
                    )
                }
                Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.SpaceBetween) {
                    Text(spTime(shownPositionMs), style = MaterialTheme.typography.labelMedium)
                    Text(spTime(s.durationMs), style = MaterialTheme.typography.labelMedium)
                }

                if (landscape) Spacer(Modifier.height(8.dp)) else Spacer(Modifier.weight(1f))

                Row(
                    if (landscape) Modifier.fillMaxWidth().padding(top = 8.dp)
                    else Modifier.fillMaxWidth().padding(top = 16.dp, bottom = 80.dp),
                    horizontalArrangement = Arrangement.SpaceEvenly,
                    verticalAlignment = Alignment.CenterVertically,
                ) {
                    IconButton(onClick = remote::previous, modifier = Modifier.size(60.dp)) {
                        SkipIcon(next = false, modifier = Modifier.size(36.dp))
                    }
                    FilledIconButton(
                        onClick = { if (s.isPaused) remote.resume() else remote.pause() },
                        modifier = Modifier.size(86.dp),
                    ) { PlayPauseIcon(!s.isPaused, Modifier.size(48.dp)) }
                    IconButton(onClick = remote::next, modifier = Modifier.size(60.dp)) {
                        SkipIcon(next = true, modifier = Modifier.size(36.dp))
                    }
                }
            }
        }
    }

    if (showDetails) {
        AlertDialog(
            onDismissRequest = { showDetails = false },
            confirmButton = { TextButton(onClick = { showDetails = false }) { Text("Закрыть") } },
            title = { Text(s.title ?: "") },
            text = { Text("Исполнитель: ${s.artist ?: "—"}\nИсточник: Spotify") },
        )
    }
}

private fun spTime(ms: Long): String {
    val total = ms / 1000
    return "${total / 60}:${(total % 60).toString().padStart(2, '0')}"

@Composable
private fun SpotifyHome(remote: SpotifyRemote, s: SpotifyState, onBeforePlay: () -> Unit) {
    val ui = LocalSpotifyMini.current
    LaunchedEffect(Unit) { if (s.shelves == null) remote.loadRecommended() }

    val play: (SpotifyItem) -> Unit = { item ->
        onBeforePlay(); ui.dismissedUri = null; remote.playUri(item.uri)
    }
    val tap: (SpotifyItem) -> Unit = { item ->
        if (item.playable) play(item) else remote.openItem(item)
    }
    val pad = PaddingValues(horizontal = 16.dp)

    LazyColumn(Modifier.fillMaxSize(), contentPadding = PaddingValues(bottom = 16.dp)) {
        item(key = "header") {
            SpotifyHeader("Музыку по ссылке можно запустить во вкладке «Поиск»: кнопка «+», затем Spotify.")
        }

        s.error?.let { message ->
            item(key = "error") {
                Text(message, Modifier.padding(horizontal = 16.dp, vertical = 4.dp), color = MaterialTheme.colorScheme.error)
            }
        }

        val browse = s.browse
        if (browse != null) {
            // Открыта внутренняя подборка: показываем её вместо рядов
            item(key = "browse-head") {
                Row(
                    Modifier.fillMaxWidth().padding(start = 16.dp, end = 8.dp, top = 20.dp, bottom = 8.dp),
                    verticalAlignment = Alignment.CenterVertically,
                ) {
                    Text(s.browseTitle ?: "Подборка", style = MaterialTheme.typography.titleLarge, modifier = Modifier.weight(1f))
                    TextButton(onClick = remote::closeBrowse) { Text("← К подборкам") }
                }
            }
            item(key = "browse-row") {
                SpotifyTileRow(browse, s.covers, contentPadding = pad, onOpenSpotify = remote::openSpotifyApp, onClick = tap)
            }
        } else {
            val recent = s.history
            if (recent.isNotEmpty()) {
                item(key = "recent") {
                    Column {
                        SpotifySectionTitle("Недавно играло")
                        SpotifyTileRow(recent, s.covers, contentPadding = pad, onOpenSpotify = remote::openSpotifyApp, onClick = play)
                    }
                }
            }

            item(key = "shelves-title") { SpotifySectionTitle("Подборки для вас") }
            val shelves = s.shelves
            when {
                shelves == null -> item(key = "loading") {
                    LinearProgressIndicator(Modifier.fillMaxWidth().padding(horizontal = 16.dp))
                }
                shelves.isEmpty() -> item(key = "empty") {
                    Column(Modifier.padding(horizontal = 16.dp)) {
                        Text("Spotify не вернул подборки. Откройте приложение Spotify, включите что-нибудь и повторите.")
                        TextButton(onClick = remote::loadRecommended) { Text("Повторить") }
                    }
                }
                else -> shelves.forEachIndexed { i, shelf ->
                    item(key = "shelf:$i") {
                        Column {
                            SpotifyRowLabel(shelf.title)
                            SpotifyTileRow(shelf.items, s.covers, contentPadding = pad, onOpenSpotify = remote::openSpotifyApp, onClick = tap)
                            Spacer(Modifier.height(12.dp))
                        }
                    }
                }
            }
        }

        item(key = "footer") {
            Row(
                Modifier.fillMaxWidth().padding(horizontal = 16.dp, vertical = 12.dp),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically,
            ) {
                SpotifyOpenButton(remote::openSpotifyApp)
                TextButton(onClick = remote::disconnect) { Text("Отключить Spotify") }
            }
        }
    }
}

@Composable
fun TrackListScreen(
    providers: List<MusicProvider>,
    audio: AudioBackend,
    session: YouTubeSession? = null,
    onOpenSpotify: () -> Unit = {},
) {
    var query by remember { mutableStateOf("") }
    var selected by remember { mutableStateOf(Service.ALL) }
    var ytQuery by remember { mutableStateOf("") }
    var retry by remember { mutableStateOf(0) }
    var results by remember { mutableStateOf<Map<ProviderId, SourceResult>>(emptyMap()) }
    var loading by remember { mutableStateOf(false) }
    val playback by audio.state.collectAsState()
    val hub = remember(providers) { MusicHub(providers) }

    // ALL ищет во всех подключённых источниках, остальные чипы только в своём
    val sourceId = ProviderId.entries.firstOrNull { it.name == selected.name }
    val canSearch = selected == Service.ALL ||
        (selected.ready && providers.any { it.id == sourceId })

    LaunchedEffect(query, selected, retry, hub) {
        if (!canSearch) {
            results = emptyMap()
            loading = false
            return@LaunchedEffect
        }
        delay(400)
        loading = true
        val fresh = mutableMapOf<ProviderId, SourceResult>()
        try {
            // «Все» ищет только там, где играют полные треки; превью Deezer живут на своём чипе
            val sources = sourceId?.let { setOf(it) }
                ?: providers.filter { p ->
                    Service.entries.firstOrNull { it.name == p.id.name }?.kind ==
                        com.unknowstorm.musicapp.model.PlaybackKind.IN_APP
                }.map { it.id }.toSet()
            hub.search(query, sources).collect { result ->
                fresh[result.provider] = result
                results = fresh.toMap() // новые треки появляются, как только ответил любой источник
            }
        } finally {
            loading = false
        }
    }

    val visible = interleave(providers.mapNotNull { results[it.id]?.tracks })
        .filterNot { DeadTracks.isDead(it) }
    val failed = providers.mapNotNull { p -> results[p.id]?.takeIf { it.error != null } }

    Column(Modifier.fillMaxSize().statusBarsPadding()) {
        if (selected != Service.SPOTIFY) OutlinedTextField(
            value = query,
            onValueChange = { query = it },
            label = { Text("Поиск") },
            singleLine = true,
            keyboardOptions = KeyboardOptions(imeAction = ImeAction.Search),
            keyboardActions = KeyboardActions(
                onSearch = { if (selected == Service.YOUTUBE) ytQuery = query.trim() },
            ),
            modifier = Modifier.fillMaxWidth().padding(16.dp)
        )

        TopBar(selected, { selected = it }, Modifier.padding(horizontal = 16.dp, vertical = 8.dp))

        if (loading) {
            LinearProgressIndicator(Modifier.fillMaxWidth())
        }

        if (!canSearch && selected != Service.SPOTIFY && !(selected == Service.YOUTUBE && session != null)) {
            Text(
                text = if (selected == Service.YOUTUBE) "YouTube ищется на своём экране: Главная → YouTube." else "${selected.title} скоро появится. ${selected.note}.",
                color = MaterialTheme.colorScheme.onSurfaceVariant,
                modifier = Modifier.padding(16.dp)
            )
        }

        // Ошибка одного источника не мешает остальным
        failed.forEach { result ->
            Row(
                Modifier.padding(horizontal = 16.dp),
                verticalAlignment = Alignment.CenterVertically,
            ) {
                Text(
                    text = "${result.provider.toService()?.title ?: result.provider.name}: ${result.error}",
                    color = MaterialTheme.colorScheme.error,
                    style = MaterialTheme.typography.bodySmall,
                    modifier = Modifier.weight(1f),
                )
                TextButton(onClick = { retry++ }) { Text("Повторить") }
            }
        }

        playback.error?.let {
            Text(
                text = it,
                color = MaterialTheme.colorScheme.error,
                modifier = Modifier.padding(horizontal = 16.dp, vertical = 8.dp)
            )
        }

        if (canSearch && !loading && results.isNotEmpty() && visible.isEmpty() && failed.isEmpty()) {
            Text(
                text = "Ничего не найдено",
                modifier = Modifier.padding(horizontal = 16.dp, vertical = 8.dp)
            )
        }

        if (selected == Service.SPOTIFY) {
            // Выдача Spotify отдельная: с другими источниками не смешивается
            SpotifySearchPane(
                remote = com.unknowstorm.musicapp.spotify.LocalSpotifyRemote.current,
                onBeforePlay = { audio.stop(); session?.close() },
                onOpenSection = onOpenSpotify,
                modifier = Modifier.weight(1f),
            )
            return@Column
        }

        if (selected == Service.YOUTUBE && session != null) {
            // Выдача YouTube отдельная: правила запрещают смешивать её с другими источниками
            YouTubeSearchPane(
                submitted = ytQuery,
                audio = audio,
                session = session,
                onLeave = { selected = Service.ALL },
                modifier = Modifier.weight(1f),
            )
            return@Column
        }

        LazyColumn(Modifier.weight(1f)) {
            items(visible, key = { "${it.provider}:${it.id}" }) { track ->
                ListItem(
                    leadingContent = {
                        TrackCover(track, Modifier.size(52.dp), badge = true, badgeSize = 18.dp)
                    },
                    headlineContent = { Text(track.title, maxLines = 1) },
                    supportingContent = { Text(track.artist, maxLines = 1) },
                    trailingContent = { Text(formatDuration(track.durationMs)) },
                    colors = ListItemDefaults.colors(
                        containerColor = if (playback.mediaId == track.id && playback.provider == track.provider) {
                            MaterialTheme.colorScheme.secondaryContainer
                        } else {
                            MaterialTheme.colorScheme.surface
                        }
                    ),
                    modifier = Modifier.clickable {
                        val playable = visible.filter { !it.streamUrl.isNullOrBlank() }
                        val index = playable.indexOfFirst {
                            it.id == track.id && it.provider == track.provider
                        }
                        if (index >= 0) audio.playQueue(playable, index)
                    }
                )
            }
        }
    }
}

private fun formatDuration(ms: Long): String {
    val totalSeconds = ms / 1000
    val minutes = totalSeconds / 60
    val seconds = totalSeconds % 60
    return "$minutes:${seconds.toString().padStart(2, '0')}"
}

/** Запрос подборки с повторами: на холодном старте сеть часто ещё не готова (DNS, таймаут). */
private suspend fun searchWithRetry(provider: MusicProvider, query: String): List<Track> {
    var last: Exception? = null
    repeat(3) { attempt ->
        try {
            return provider.search(query)
        } catch (e: CancellationException) {
            throw e
        } catch (e: Exception) {
            last = e
            delay(800L * (attempt + 1))
        }
    }
    throw last ?: IllegalStateException("search failed")
}

/** Лента одного источника. Живёт на уровне MainScreen, чтобы не грузиться заново при смене вкладок. */
class HomeFeed(
    val service: Service,
    private val provider: MusicProvider,
    private val mixes: List<Pair<String, String>>,
) {
    var sections by mutableStateOf<List<FeedSection>>(emptyList())
        private set
    var loading by mutableStateOf(false)
        private set
    var error by mutableStateOf<String?>(null)
        private set

    suspend fun loadIfNeeded() {
        if (sections.isEmpty() && !loading) {
            load()
            // Если сеть так и не ответила, пробуем ещё пару раз сами, без нажатия «Повторить»
            var tries = 0
            while (sections.isEmpty() && error != null && tries < 2) {
                tries++
                delay(4000)
                load()
            }
        }
    }

    suspend fun load() {
        loading = true
        error = null
        try {
            var lastError: String? = null
            val loaded = coroutineScope {
                mixes.map { (title, query) ->
                    async {
                        try {
                            FeedSection(title, searchWithRetry(provider, query).take(12))
                        } catch (e: CancellationException) {
                            throw e
                        } catch (e: Exception) {
                            println("Solara: feed ${service.title}/$title failed: ${e.stackTraceToString()}")
                            lastError = e.message
                            null
                        }
                    }
                }.awaitAll()
            }.filterNotNull().filter { it.tracks.isNotEmpty() }
            sections = loaded
            if (loaded.isEmpty() && lastError != null) error = "Ошибка загрузки: $lastError"
        } finally {
            loading = false
        }
    }
}

private data class FeedRow(val key: String, val title: String, val service: Service?, val tracks: List<Track>)

private fun <T> roundRobin(lists: List<List<T>>): List<T> {
    val result = mutableListOf<T>()
    val longest = lists.maxOfOrNull { it.size } ?: 0
    for (i in 0 until longest) {
        for (list in lists) list.getOrNull(i)?.let(result::add)
    }
    return result
}

/** Собирает ряды главной. На вкладке «Все» источники чередуются, а сверху идёт общий микс. */
private fun buildRows(feeds: List<HomeFeed>, mixed: Boolean): List<FeedRow> {
    val perFeed = feeds.map { feed ->
        feed.sections.map { FeedRow("${feed.service.name}:${it.title}", it.title, feed.service, it.tracks.filterNot { t -> DeadTracks.isDead(t) }) }
    }
    val rows = roundRobin(perFeed)
    if (!mixed) return rows
    val firstRows = perFeed.mapNotNull { it.firstOrNull() }
    if (firstRows.size < 2) return rows
    val mix = roundRobin(firstRows.map { it.tracks }).take(16)
    return listOf(FeedRow("mix", "Микс из всех источников", Service.ALL, mix)) + rows
}

/**
 * Лайки хранят только метаданные, поэтому ссылку на звук берём заново у источника:
 * у Deezer она протухает. Что не удалось получить, пропускаем.
 */
internal suspend fun playLiked(
    tracks: List<Track>,
    start: Track,
    providers: List<MusicProvider>,
    audio: AudioBackend,
) {
    // Окно вокруг выбранного трека: так работает и выбор из хвоста длинного списка
    val from = tracks.indexOfFirst { it.provider == start.provider && it.id == start.id }.coerceAtLeast(0)
    val window = tracks.drop((from - 5).coerceAtLeast(0)).take(30)
    val resolved = coroutineScope {
        window.map { track ->
            async {
                val provider = providers.firstOrNull { it.id == track.provider } ?: return@async null
                val url = try {
                    withTimeoutOrNull(6_000) { provider.resolveStream(track) }
                } catch (e: CancellationException) {
                    throw e
                } catch (e: Exception) {
                    null
                }
                url?.takeIf { it.isNotBlank() }?.let { track.copy(streamUrl = it) }
            }
        }.awaitAll()
    }.filterNotNull()
    val index = resolved.indexOfFirst { it.provider == start.provider && it.id == start.id }
    if (index >= 0) audio.playQueue(resolved, index)
}

package com.unknowstorm.musicapp.ui

import com.unknowstorm.musicapp.model.Track
import com.unknowstorm.musicapp.provider.MusicProvider
import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.async
import kotlinx.coroutines.awaitAll
import kotlinx.coroutines.coroutineScope

/** Ищет во всех источниках сразу и перемешивает результаты. Ошибка одного источника не ломает остальные. */
suspend fun searchAll(providers: List<MusicProvider>, query: String): List<Track> {
    var firstError: Exception? = null
    var failures = 0
    val lists = coroutineScope {
        providers.map { provider ->
            async {
                try {
                    provider.search(query)
                } catch (e: CancellationException) {
                    throw e
                } catch (e: Exception) {
                    println("Solara: ${provider.id} search failed: ${e.message}")
                    failures++
                    firstError = firstError ?: e
                    emptyList<Track>()
                }
            }
        }.awaitAll()
    }
    if (providers.isNotEmpty() && failures == providers.size) {
        throw firstError ?: IllegalStateException("Источники недоступны")
    }
    return interleave(lists)
}

private fun <T> interleave(lists: List<List<T>>): List<T> {
    val result = mutableListOf<T>()
    val longest = lists.maxOfOrNull { it.size } ?: 0
    for (i in 0 until longest) {
        for (list in lists) list.getOrNull(i)?.let(result::add)
    }
    return result
}
